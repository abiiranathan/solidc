66size_t chan_pubsub_subscribe(chan_pubsub_t* ps) {
67 if (!ps) return CHAN_PUBSUB_NO_SUB;
68
69 size_t id = CHAN_PUBSUB_NO_SUB;
70 lock_acquire(&ps->lock);
71 if (!ps->closed && ps->sub_count < ps->max_subs) {
72 for (size_t i = 0; i < ps->max_subs; i++) {
73 if (!ps->subs[i].active) {
74 /* Create the queue outside the lock to keep publish()
75 * contention low; mark the slot first so nobody else takes
76 * it. */
77 ps->subs[i].active = true;
78 id = i;
79 break;
80 }
81 }
82 if (id != CHAN_PUBSUB_NO_SUB) {
83 /* Pre-increment so concurrent subscribe() calls pick other
84 * slots while this one builds its queue. */
85 ps->sub_count++;
86 }
87 }
88 lock_release(&ps->lock);
89
90 if (id == CHAN_PUBSUB_NO_SUB) return CHAN_PUBSUB_NO_SUB;
91
92 Channel* q = chan_new(ps->value_size, ps->queue_depth);
93 if (!q) {
94 /* Roll the reservation back. */
95 lock_acquire(&ps->lock);
96 ps->subs[id].active = false;
97 ps->sub_count--;
98 lock_release(&ps->lock);
99 return CHAN_PUBSUB_NO_SUB;
100 }
101
102 lock_acquire(&ps->lock);
103 ps->subs[id].queue = q;
104 lock_release(&ps->lock);
105 return id;
106}

108ChanStatus chan_pubsub_publish(chan_pubsub_t* ps, const void* value) {
109 if (!ps || !value) return CHAN_INVALID;
110
111 ChanStatus overall = CHAN_OK;
112 lock_acquire(&ps->lock);
113 if (ps->closed) {
114 lock_release(&ps->lock);
115 return CHAN_CLOSED;
116 }
117 for (size_t i = 0; i < ps->max_subs; i++) {
118 if (!ps->subs[i].active) continue;
119 ChanStatus st = chan_try_send(ps->subs[i].queue, value);
120 if (st != CHAN_OK && overall == CHAN_OK) overall = st;
121 }
122 lock_release(&ps->lock);
123 return overall;
124}

134ChanStatus chan_pubsub_unsubscribe(chan_pubsub_t* ps, size_t id) {
135 if (!ps || id >= ps->max_subs) return CHAN_INVALID;
136
137 lock_acquire(&ps->lock);

139 lock_release(&ps->lock);
140 return CHAN_INVALID;

196struct chan_workers {
197 Channel* work; /* shared work channel (values) */
198 chan_work_fn fn; /* user callback */
199 void* user; /* user context */
200 size_t value_size;
201
202 Thread* threads; /* worker_count handles */
203 size_t worker_count;
204
205 /* Pending = submitted and not yet completed. Workers debit the shared
206 * counter after their callback returns, so shutdown() can drain
207 * precisely. */
208 Lock pending_lock;
209 size_t pending;
210
211 bool shutdown;

216static void* chan_worker_main(void* arg) {
217 chan_workers_t* w = (chan_workers_t*)arg;
218
219 void* value = malloc(w->value_size);
220 if (!value) return NULL;
221
222 for (;;) {
223 if (chan_recv(w->work, value)) {
224 w->fn(value, w->user);
225 lock_acquire(&w->pending_lock);
226 w->pending--;
227 lock_release(&w->pending_lock);
228 } else {
229 break; /* channel closed and drained */
230 }
231 }
232
233 free(value);
234 return NULL;
235}

340int chan_select(Channel** channels, void** outs, size_t count, int timeout_ms) {
341 if (!channels || count == 0 || count > CHAN_SELECT_MAX) return CHAN_SELECT_TIMEOUT;
342
343 /* When outs is NULL the winning value is discarded: receive into a
344 * scratch buffer sized for the largest channel. */
345 void* scratch = NULL;
346 if (!outs) {
347 size_t max_sz = 0;
348 for (size_t i = 0; i < count; i++) {
349 size_t sz = chan_value_size(channels[i]);
350 if (sz > max_sz) max_sz = sz;
351 }
352 if (max_sz == 0) max_sz = 1;
353 scratch = malloc(max_sz);
354 if (!scratch) return CHAN_SELECT_TIMEOUT;
355 }
356
357 int result = CHAN_SELECT_TIMEOUT;
358
359 /* Phase 1: one immediate sweep in priority order. */
360 for (size_t i = 0; i < count; i++) {
361 void* target = outs ? outs[i] : scratch;
362 ChanStatus st = chan_try_recv(channels[i], target);
363 if (st == CHAN_OK) {
364 result = (int)i;
365 goto out;
366 }
367 }
368
369 if (timeout_ms == 0) goto out;
370
371 /* Phase 2: poll with an adaptive backoff until the budget expires. */
372 int waited = 0;
373 int slice = 1; /* ms; doubles up to 16 */
374 while (timeout_ms < 0 || waited < timeout_ms) {
375 int wait = (timeout_ms < 0) ? slice : ((slice < timeout_ms - waited) ? slice : timeout_ms - waited);
376 NANOSLEEP(0, (long)wait * 1000000L);
377 waited += wait;
378 if (slice < 16) slice *= 2;
379
380 for (size_t i = 0; i < count; i++) {
381 void* target = outs ? outs[i] : scratch;
382 ChanStatus st = chan_try_recv(channels[i], target);
383 if (st == CHAN_OK) {
384 result = (int)i;
385 goto out;
386 }
387 }
388 }
389
390out:
391 free(scratch);
392 return result;
393}

417static void* chan_merger_main(void* arg) {
418 merger_arg* m = (merger_arg*)arg;
419 chan_merge_ctx* ctx = m->ctx;
420
421 void* value = malloc(m->value_size);
422 if (value) {
423 while (chan_recv(m->src, value)) {
424 /* Forward; if dest fails (OOM growth), drop the value rather
425 * than deadlock the source. */
426 chan_send(m->dest, value);
427 }
428 free(value);
429 }
430
431 /* Last forwarder out: free the shared context and close dest. */
432 if (atomic_fetch_sub(&ctx->live, 1) == 1) {
433 chan_close(m->dest);
434 free(ctx->args);
435 free(ctx);
436 }
437 return NULL;
438}

440ChanStatus chan_merge(Channel* dest, Channel** sources, size_t count) {
441 if (!dest || !sources || count == 0 || count > CHAN_SELECT_MAX) return CHAN_INVALID;
442 for (size_t i = 0; i < count; i++) {
443 if (!sources[i]) return CHAN_INVALID;
444 if (chan_value_size(sources[i]) != chan_value_size(dest)) return CHAN_INVALID;
445 }
446
447 chan_merge_ctx* ctx = (chan_merge_ctx*)calloc(1, sizeof(chan_merge_ctx));
448 if (!ctx) return CHAN_NOMEM;
449 ctx->args = (merger_arg*)calloc(count, sizeof(merger_arg));
450
451 atomic_store(&ctx->live, count);
452
453 Thread* threads = (Thread*)calloc(count, sizeof(Thread));
454 if (!ctx->args || !threads) {
455 free(ctx->args);
456 free(threads);
457 free(ctx);
458 return CHAN_NOMEM;
459 }
460
461 size_t started = 0;
462 for (; started < count; started++) {
463 ctx->args[started].dest = dest;
464 ctx->args[started].src = sources[started];
465 ctx->args[started].value_size = chan_value_size(dest);
466 ctx->args[started].ctx = ctx;
467 if (thread_create(&threads[started], chan_merger_main, &ctx->args[started]) != 0) break;
468 /* Forwarders are fire-and-forget: detached so they reclaim
469 * themselves when the source closes (no join API needed). */
470 thread_detach(threads[started]);
471 }
472
473 if (started < count) {
474 if (started == 0) {
475 /* No forwarder running: nobody will free the context. */
476 free(ctx->args);
477 free(ctx);
478 free(threads);
479 return CHAN_NOMEM;
480 }
481 /* Partial start: close the sources with running forwarders so they
482 * exit; the last one frees the context and closes dest. Slots that
483 * never started must not hold the counter open. */
484 atomic_fetch_sub(&ctx->live, (int)(count - started));
485 for (size_t i = 0; i < started; i++) chan_close(sources[i]);
486 for (size_t i = 0; i < started; i++) {
487 void* ret;
488 thread_join(threads[i], &ret);
489 }
490 free(threads);
491 return CHAN_NOMEM;
492 }
493
494 free(threads); /* forwarders are self-managing from here */
495 return CHAN_OK;
496}

71/*
72 * Returns the extension of the BASENAME, including the dot, or "" if the
73 * basename has none. Two subtleties handled here:
74 * - The search must start AFTER the last separator, otherwise
75 * "/path/to.dir/file" would report ".dir/file".
76 * - A leading dot in the basename marks a hidden file ("~/.bashrc"),
77 * not an extension, so ".bashrc" yields "".
78 * Multi-part names behave as expected: "archive.tar.gz" -> ".gz".
79 */
82void filepath_extension(const char* path, char* ext, size_t size) {
83 if (!path || !ext || size == 0) {
84 if (ext) ext[0] = '\0';
85 return;
86 }
87
88 const char* base = last_separator(path);
89 base = base ? base + 1 : path;
90
91 const char* dot = strrchr(base, '.');
92 if (!dot || dot == base) {
93 ext[0] = '\0';
94 return;
95 }
96 fp_strlcpy(ext, dot, size);
97}

148bool filepath_join_buf(const char* path1, const char* path2, char* abspath, size_t len) {
149 if (!path1 || !path2 || !abspath || len == 0) {
150 if (abspath && len > 0) abspath[0] = '\0';
151 errno = EINVAL;
152 return false;
153 }
154
155#ifdef _WIN32
156 // On Windows, decide which separator to use based on what path1 already uses
157 const char* sep = strchr(path1, '\\') ? "\\" : "/";
158 int result = snprintf(abspath, len, "%s%s%s", path1, sep, path2);
159#else
160 int result = snprintf(abspath, len, "%s/%s", path1, path2);
161#endif
162
163 // Check for encoding error or truncation
164 if (result < 0 || (size_t)result >= len) {
165 // Ensure null-termination if truncated (snprintf does this, but good practice to be sure)
166 abspath[len - 1] = '\0';
167 errno = ENAMETOOLONG;
168 return false;
169 }
170
171 return true;
172}

125ProcessError process_run_and_capture(const char* command, const char* const argv[], ProcessOptions* options,
126 int* exit_code) {
127 ProcessHandle* proc = NULL;
128 ProcessError err = {0};
129 err = process_create(&proc, command, argv, options);
130 if (err != PROCESS_SUCCESS) {
131 return err;
132 }
133
134 ProcessResult res = {0};
135 err = process_wait(proc, &res, -1);
136 process_free(proc);
137
138 if (exit_code) {
139 *exit_code = res.exit_code;
140 }
141
142 if (err != PROCESS_SUCCESS) {
143 return err;
144 }
145
146 if (res.exit_code != 0) {
147 return PROCESS_ERROR_EXEC_FAILED;
148 }
149 return PROCESS_SUCCESS;
150}

535/* =========================================================================
536 * TIMING AND PERFORMANCE
537 *
538 * Two families of macros:
539 *
540 * TIME_BLOCK(name, block) — measure a block and print elapsed seconds.
541 * TIME_BLOCK_MS(name, block) — measure a block and print elapsed milliseconds.
542 *
543 * `block` is an unquoted sequence of statements placed verbatim before the
544 * stop-time call, e.g.:
545 *
546 * TIME_BLOCK("sort", sort(arr, n);)
547 *
548 * get_time_ns() / get_time_ms() — return current monotonic time as uint64_t.
549 * TIME_DIFF / TIME_DIFF_MS — compute elapsed seconds / milliseconds between
550 * two timestamps captured with the platform-specific API.
551 * ========================================================================= */
552
553#if defined(_WIN32)
554
555 #include "platform.h" /* LARGE_INTEGER, QueryPerformanceCounter, etc. */
556
561 #define TIME_DIFF(start, end, freq) ((double)((end).QuadPart - (start).QuadPart) / (double)(freq).QuadPart)
562
567 #define TIME_DIFF_MS(start, end, freq) \
568 ((double)((end).QuadPart - (start).QuadPart) * 1000.0 / (double)(freq).QuadPart)
569
570 #define TIME_BLOCK(name, block) \
571 do { \
572 LARGE_INTEGER _freq, _t0, _t1; \
573 QueryPerformanceFrequency(&_freq); \
574 QueryPerformanceCounter(&_t0); \
575 block QueryPerformanceCounter(&_t1); \

577 } while (0)

630static inline uint64_t get_time_ns(void) {
631#if defined(_WIN32)
632 LARGE_INTEGER freq, counter;
633 QueryPerformanceFrequency(&freq);
634 QueryPerformanceCounter(&counter);
635 /* Split into whole seconds and remainder to avoid 64-bit overflow. */
636 uint64_t seconds = (uint64_t)counter.QuadPart / (uint64_t)freq.QuadPart;
637 uint64_t remainder = (uint64_t)counter.QuadPart % (uint64_t)freq.QuadPart;
638 return seconds * UINT64_C(1000000000) + (remainder * UINT64_C(1000000000)) / (uint64_t)freq.QuadPart;
639#else
640 struct timespec ts;
641 clock_gettime(CLOCK_MONOTONIC, &ts);
642 return (uint64_t)ts.tv_sec * UINT64_C(1000000000) + (uint64_t)ts.tv_nsec;
643#endif
644}

654static inline uint64_t get_time_ms(void) { return get_time_ns() / UINT64_C(1000000); }

790/* =========================================================================
791 * SIMD FEATURE DETECTION
792 *
793 * Two layers are provided:
794 *
795 * Compile-time (HAS_* macros, usable in #if): reflect the instruction
796 * set the *compiler* was told to target (e.g. via -mavx2 or /arch:AVX2).
797 * These say nothing about the CPU the binary will actually run on.
798 *
799 * Runtime (cpu_has_* functions): query the actual CPU at execution time.
800 * Always check the runtime function before executing the corresponding
801 * instructions, even if the matching HAS_* macro is 1 — the binary may
802 * be run on different hardware than it was compiled for (e.g. a binary
803 * built with -mavx2 but distributed and run on an older CPU lacking it,
804 * which will SIGILL on the first AVX2 instruction).
805 *
806 * x86/x64: runtime detection uses __builtin_cpu_supports() on GCC/Clang
807 * (backed by glibc's IFUNC CPUID parsing) and raw CPUID via __cpuid /

812 * Linux uses getauxval(AT_HWCAP) & HWCAP_NEON. Non-Linux ARM32 targets
813 * fall back to the compile-time result.

1151/* -------------------------------------------------------------------------

1153 *
1154 * Uses the strongest primitive available per platform:
1155 * - memset_s() C11 Annex K (glibc with __STDC_WANT_LIB_EXT1__)
1156 * - SecureZeroMemory() Win32 (explicit volatile implementation)

1159 * ---------------------------------------------------------------------- */

1167#if defined(__STDC_LIB_EXT1__) || defined(__STDC_WANT_LIB_EXT1__)
1168 memset_s(p, n, 0, n);

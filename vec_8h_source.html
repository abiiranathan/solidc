375static inline SimdVec2 vec2_rotate(SimdVec2 v, float angle) {
376 float c = cosf(angle);
377 float s = sinf(angle);
378
379 // Broadcast components to all lanes for vectorized linear combination
380 simd_vec_t x = simd_splat_x(v.v);
381 simd_vec_t y = simd_splat_y(v.v);
382
383 // Rotation matrix columns:
384 // col0 = {c, s, 0, 0} - maps X component
385 // col1 = {-s, c, 0, 0} - maps Y component
386 simd_vec_t c0 = simd_set(c, s, 0.0f, 0.0f);
387 simd_vec_t c1 = simd_set(-s, c, 0.0f, 0.0f);
388
389 // result = x * col0 + y * col1
390 return (SimdVec2){.v = simd_add(simd_mul(x, c0), simd_mul(y, c1))};
391}

716static inline SimdVec3 vec3_perpendicular(SimdVec3 v) {
717 // Strategy: Cross v with the world axis it is LEAST parallel to.
718 // If |v.y| < 0.9, cross with Y-axis (0,1,0).
719 // Otherwise cross with X-axis (1,0,0).
720
721 // We access scalars here because conditional logic is cleaner in scalar
722 // than trying to construct a branchless SIMD mask for this specific logic.
723 SimdVec3 axis;
724 if (fabsf(v.y) < 0.99f) {
725 axis = vec3_load((Vec3){0.0f, 1.0f, 0.0f}); // Up
726 } else {
727 axis = vec3_load((Vec3){1.0f, 0.0f, 0.0f}); // Right
728 }
729
730 return vec3_normalize(vec3_cross(v, axis));
731}

1332static inline Vec3 vec3_barycentric(SimdVec3 a, SimdVec3 b, SimdVec3 c, SimdVec3 p) {
1333 SimdVec3 v0 = vec3_sub(b, a);
1334 SimdVec3 v1 = vec3_sub(c, a);
1335 SimdVec3 v2 = vec3_sub(p, a);
1336
1337 float d00 = vec3_dot(v0, v0);
1338 float d01 = vec3_dot(v0, v1);
1339 float d11 = vec3_dot(v1, v1);
1340 float d20 = vec3_dot(v2, v0);
1341 float d21 = vec3_dot(v2, v1);
1342
1343 float denom = d00 * d11 - d01 * d01;
1344 if (fabsf(denom) < 1e-12f) {
1345 // Degenerate triangle: weights are meaningless; return centroid weight.
1346 return (Vec3){1.0f / 3.0f, 1.0f / 3.0f, 1.0f / 3.0f};
1347 }
1348
1349 float v = (d11 * d20 - d01 * d21) / denom;
1350 float w = (d00 * d21 - d01 * d20) / denom;
1351 float u = 1.0f - v - w;
1352 return (Vec3){u, v, w};
1353}

1365static inline SimdVec3 vec3_refract(SimdVec3 i, SimdVec3 n, float eta) {
1366 float cosi = -vec3_dot(i, n);
1367 float k = 1.0f - eta * eta * (1.0f - cosi * cosi);
1368 if (k < 0.0f) {
1369 return (SimdVec3){{0}}; // Total internal reflection
1370 }
1371 return vec3_add(vec3_mul(i, eta), vec3_mul(n, eta * cosi - sqrtf(k)));

318Flag* flag_add(FlagParser* fp, FlagDataType type, const char* name, char short_name,
319 const char* desc, void* value_ptr, bool required) {
320 if (!fp || !name || !value_ptr) return NULL;

322 if (fp->flag_count >= fp->flag_capacity) {

335 f->name = arena_strdup(fp->arena, name);

338 f->value_ptr = value_ptr;

341 f->persistent = false;
342 f->validator = NULL;

344 // Copy the default value for display in help
345 f->default_ptr = copy_default_value(fp->arena, type, value_ptr);
346 return f;

641static FlagStatus parse_value(FlagParser* fp, Flag* flag, const char* str) {
642 if (!str) return FLAG_ERROR_MISSING_VALUE;
643 char* endptr = NULL;
644 errno = 0;
645
646 switch (flag->type) {
647 case TYPE_BOOL: {
648 bool val = false;
649 if (!parse_bool_value(str, &val)) return FLAG_ERROR_INVALID_ARGUMENT;
650 *(bool*)flag->value_ptr = val;
651 break;
652 }
653 case TYPE_CHAR:
654 if (!str || strlen(str) == 0) return FLAG_ERROR_MISSING_VALUE;
655 if (strlen(str) != 1) return FLAG_ERROR_INVALID_ARGUMENT;
656 *(char*)flag->value_ptr = str[0];
657 break;
658 case TYPE_STRING:
659 // Don't free the old value - we don't know if it's malloc'd or static,
660 // and we are moving to arena-managed strings.
661 // if (*(char**)flag->value_ptr) free(*(char**)flag->value_ptr);
662 *(char**)flag->value_ptr = arena_strdup(fp->arena, str);
663 break;
664
665 // Signed Integers
666 case TYPE_INT8:
667 case TYPE_INT16:
668 case TYPE_INT32:
669 case TYPE_INT64: {
670 long long val = strtoll(str, &endptr, 10);
671 if (endptr == str || *endptr != '\0' || errno == ERANGE)
672 return FLAG_ERROR_INVALID_NUMBER;
673
674 if (flag->type == TYPE_INT8 && !check_range_int(val, INT8_MIN, INT8_MAX))
675 return FLAG_ERROR_INVALID_NUMBER;
676 if (flag->type == TYPE_INT16 && !check_range_int(val, INT16_MIN, INT16_MAX))
677 return FLAG_ERROR_INVALID_NUMBER;
678 if (flag->type == TYPE_INT32 && !check_range_int(val, INT32_MIN, INT32_MAX))
679 return FLAG_ERROR_INVALID_NUMBER;
680
681 // Assign based on type size
682 if (flag->type == TYPE_INT8)
683 *(int8_t*)flag->value_ptr = (int8_t)val;
684 else if (flag->type == TYPE_INT16)
685 *(int16_t*)flag->value_ptr = (int16_t)val;
686 else if (flag->type == TYPE_INT32)
687 *(int32_t*)flag->value_ptr = (int32_t)val;
688 else
689 *(int64_t*)flag->value_ptr = (int64_t)val;
690 break;
691 }
692
693 // Unsigned Integers
694 case TYPE_UINT8:
695 case TYPE_UINT16:
696 case TYPE_UINT32:
697 case TYPE_UINT64:
698 case TYPE_SIZE_T: {
699 if (str[0] == '-') return FLAG_ERROR_INVALID_NUMBER;
700 unsigned long long val = strtoull(str, &endptr, 10);
701 if (endptr == str || *endptr != '\0' || errno == ERANGE)
702 return FLAG_ERROR_INVALID_NUMBER;
703
704 if (flag->type == TYPE_UINT8 && !check_range_uint(val, UINT8_MAX))
705 return FLAG_ERROR_INVALID_NUMBER;
706 if (flag->type == TYPE_UINT16 && !check_range_uint(val, UINT16_MAX))
707 return FLAG_ERROR_INVALID_NUMBER;
708 if (flag->type == TYPE_UINT32 && !check_range_uint(val, UINT32_MAX))
709 return FLAG_ERROR_INVALID_NUMBER;
710 if (flag->type == TYPE_SIZE_T && !check_range_uint(val, SIZE_MAX))
711 return FLAG_ERROR_INVALID_NUMBER;
712
713 if (flag->type == TYPE_UINT8)
714 *(uint8_t*)flag->value_ptr = (uint8_t)val;
715 else if (flag->type == TYPE_UINT16)
716 *(uint16_t*)flag->value_ptr = (uint16_t)val;
717 else if (flag->type == TYPE_UINT32)
718 *(uint32_t*)flag->value_ptr = (uint32_t)val;
719 else if (flag->type == TYPE_SIZE_T)
720 *(size_t*)flag->value_ptr = (size_t)val;
721 else
722 *(uint64_t*)flag->value_ptr = (uint64_t)val;
723 break;
724 }
725
726 case TYPE_FLOAT: {
727 float val = strtof(str, &endptr);
728 if (endptr == str || *endptr != '\0' || errno == ERANGE)
729 return FLAG_ERROR_INVALID_NUMBER;
730 *(float*)flag->value_ptr = val;
731 break;
732 }
733 case TYPE_DOUBLE: {
734 double val = strtod(str, &endptr);
735 if (endptr == str || *endptr != '\0' || errno == ERANGE)
736 return FLAG_ERROR_INVALID_NUMBER;
737 *(double*)flag->value_ptr = val;
738 break;
739 }
740 default:
741 return FLAG_ERROR_INVALID_ARGUMENT;
742 }
743 return FLAG_OK;
744}

767FlagStatus flag_parse(FlagParser* fp, int argc, char** argv) {
768 if (!fp || argc < 1) return FLAG_ERROR_INVALID_ARGUMENT;
769
770 int i = 1;
771 bool end_of_opts = false;
772
773 while (i < argc) {
774 char* arg = argv[i];
775
776 // 1. End of Options
777 if (!end_of_opts && strcmp(arg, "--") == 0) {
778 end_of_opts = true;
779 i++;
780 continue;
781 }
782
783 // 2. Positionals (or Subcommands)
784 if (end_of_opts || arg[0] != '-') {
785 // Check subcommands
786 if (!end_of_opts && fp->cmd_count > 0) {
787 for (size_t c = 0; c < fp->cmd_count; c++) {
788 if (strcmp(fp->subcommands[c]->name, arg) == 0) {
789 fp->active_subcommand = fp->subcommands[c];
790 // Recursive call for subcommand
791 return flag_parse(fp->active_subcommand, argc - i, argv + i);
792 }
793 }
794 }
795
796 // Add positional
797 if (fp->pos_count >= fp->pos_capacity) {
798 size_t new_cap = (fp->pos_capacity == 0) ? INITIAL_CAPACITY : fp->pos_capacity * 2;
799 char** new_pos = arena_alloc(fp->arena, sizeof(char*) * new_cap);
800 if (!new_pos) {
801 fprintf(stderr, "failed to allocated new positional args");
802 exit(EXIT_FAILURE);
803 }
804
805 if (fp->positional_args) {
806 memcpy(new_pos, fp->positional_args, fp->pos_count * sizeof(char*));
807 }
808 fp->positional_args = new_pos;
809 fp->pos_capacity = new_cap;
810 }
811
812 fp->positional_args[fp->pos_count++] = arg;
813 i++;
814 continue;
815 }
816
817 // 3. Long Flags
818 if (strncmp(arg, "--", 2) == 0) {
819 char* name_start = arg + 2;
820 if (strcmp(name_start, "help") == 0) {
821 flag_print_usage(fp);
822 exit(0);
823 }
824
825 char* eq = strchr(name_start, '=');
826 char name_buf[MAX_FLAG_NAME_LEN];
827 const char* val_str = NULL;
828 const char* lookup_name = name_start;
829
830 if (eq) {
831 size_t len = (size_t)(eq - name_start);
832 if (len >= MAX_FLAG_NAME_LEN) len = MAX_FLAG_NAME_LEN - 1;
833 memcpy(name_buf, name_start, len);
834 name_buf[len] = '\0';
835 lookup_name = name_buf;
836 val_str = eq + 1;
837 }
838
839 Flag* f = find_flag_long(fp, lookup_name);
840 if (!f) {
841 set_error(fp, "Unknown flag: --%s", lookup_name);
842 return FLAG_ERROR_UNKNOWN_FLAG;
843 }
844
845 f->is_present = true;
846
847 if (f->type == TYPE_BOOL) {
848 bool b = true;
849 if (val_str && !parse_bool_value(val_str, &b)) {
850 set_error(fp, "Invalid value for --%s: '%s' (expected true/false)", f->name,
851 val_str);
852 return FLAG_ERROR_INVALID_ARGUMENT;
853 }
854 *(bool*)f->value_ptr = b;
855 } else {
856 if (!val_str) {
857 if (i + 1 < argc && is_value_token(argv[i + 1]))
858 val_str = argv[++i];
859 else {
860 set_error(fp, "Flag --%s requires a value", f->name);
861 return FLAG_ERROR_MISSING_VALUE;
862 }
863 }
864 FlagStatus s = parse_value(fp, f, val_str);
865 if (s != FLAG_OK) {
866 set_error(fp, "Invalid value for --%s: '%s' (Type mismatch or overflow)",
867 f->name, val_str);
868 return s;
869 }
870 }
871 i++;
872 continue;
873 }
874
875 // 4. Short Flags
876 if (arg[0] == '-') {
877 size_t len = strlen(arg);
878 for (size_t k = 1; k < len; k++) {
879 char c = arg[k];
880 Flag* f = find_flag_short(fp, c);
881 if (!f) {
882 set_error(fp, "Unknown short flag: -%c", c);
883 return FLAG_ERROR_UNKNOWN_FLAG;
884 }
885 f->is_present = true;
886
887 if (f->type == TYPE_BOOL) {
888 *(bool*)f->value_ptr = true;
889 } else {
890 const char* val_str = NULL;
891 if (k + 1 < len) {
892 val_str = &arg[k + 1];
893 if (*val_str == '=') val_str++;
894 FlagStatus s = parse_value(fp, f, val_str);
895 if (s != FLAG_OK) {
896 set_error(fp, "Invalid value for -%c (Type mismatch or overflow)", c);
897 return s;
898 }
899 break; // consumed rest
900 } else {
901 if (i + 1 < argc && is_value_token(argv[i + 1])) {
902 val_str = argv[++i];
903 FlagStatus s = parse_value(fp, f, val_str);
904 if (s != FLAG_OK) {
905 set_error(fp, "Invalid value for -%c (Type mismatch or overflow)",
906 c);
907 return s;
908 }
909 } else {
910 set_error(fp, "Flag -%c requires a value", c);
911 return FLAG_ERROR_MISSING_VALUE;
912 }
913 }
914 }
915 }
916 i++;
917 continue;
918 }
919 }

921 // Validation: own flags plus inherited persistent flags from ancestors.
922 // Ancestor persistent flags may have been set at this (leaf) level, and
923 // root-level validation is skipped when a subcommand is dispatched, so
924 // the leaf is responsible for enforcing them.
925 for (size_t k = 0; k < fp->flag_count; k++) {
926 Flag* f = &fp->flags[k];
927 if (f->required && !f->is_present) {
928 set_error(fp, "Missing required flag: --%s", f->name);
929 return FLAG_ERROR_REQUIRED_MISSING;
930 }
931
932 if (f->is_present && f->validator) {
933 if (f->value_ptr == NULL) {
934 set_error(fp, "Validation failed for --%s: %s", f->name, "value is NULL");
935 return FLAG_ERROR_VALIDATION;
936 }
937
938 const char* err = NULL;
939 if (!f->validator(f->value_ptr, &err)) {
940 set_error(fp, "Validation failed for --%s: %s", f->name, err ? err : "invalid");
941 return FLAG_ERROR_VALIDATION;
942 }
943 }
944 }

946 for (FlagParser* p = fp->parent; p; p = p->parent) {
947 for (size_t k = 0; k < p->flag_count; k++) {
948 Flag* f = &p->flags[k];
949 if (!f->persistent) continue;
950 if (is_shadowed_long(fp, p, f->name)) continue;
951 if (f->required && !f->is_present) {
952 set_error(fp, "Missing required flag: --%s", f->name);
953 return FLAG_ERROR_REQUIRED_MISSING;
954 }
955 if (f->is_present && f->validator) {
956 if (f->value_ptr == NULL) {
957 set_error(fp, "Validation failed for --%s: %s", f->name, "value is NULL");
958 return FLAG_ERROR_VALIDATION;
959 }
960 const char* err = NULL;
961 if (!f->validator(f->value_ptr, &err)) {
962 set_error(fp, "Validation failed for --%s: %s", f->name, err ? err : "invalid");
963 return FLAG_ERROR_VALIDATION;
964 }
965 }
966 }
967 }
968
969 return FLAG_OK;
970}

1177static void print_help_internal(FlagParser* fp) {
1178 if (!fp) return;
1179
1180 // --- 1. Header & Description ---
1181 printf("\n%s\n", fp->description ? fp->description : fp->name);
1182
1183 bool has_inherited = false;
1184 for (FlagParser* p = fp->parent; p && !has_inherited; p = p->parent) {
1185 for (size_t i = 0; i < p->flag_count; i++) {
1186 if (p->flags[i].persistent && !is_shadowed_long(fp, p, p->flags[i].name)) {
1187 has_inherited = true;
1188 break;
1189 }
1190 }
1191 }
1192
1193 printf("\nUsage:\n %s", fp->name);
1194 if (fp->flag_count > 0 || has_inherited) printf(" [flags]");
1195 if (fp->cmd_count > 0) printf(" [command]");
1196 printf("\n");
1197
1198 size_t width = calculate_display_width(fp);
1199 if (width < 20) width = 20;
1200
1201 // Local (non-persistent) flags defined on this node ---
1202 bool printed_local_header = false;
1203 for (size_t i = 0; i < fp->flag_count; i++) {
1204 if (fp->flags[i].persistent) continue;
1205 if (!printed_local_header) {
1206 printf("\nFlags:\n");
1207 printed_local_header = true;
1208 }
1209 print_flag_row(&fp->flags[i], width);
1210 }
1211
1212 // Persistent flags defined on this node (inherited by children) ---
1213 bool printed_persistent_header = false;
1214 for (size_t i = 0; i < fp->flag_count; i++) {
1215 if (!fp->flags[i].persistent) continue;
1216 if (!printed_persistent_header) {
1217 printf("\nPersistent Flags:\n");
1218 printed_persistent_header = true;
1219 }
1220 print_flag_row(&fp->flags[i], width);
1221 }
1222
1223 // Persistent flags inherited from ancestors ---
1224 if (has_inherited) {
1225 printf("\nInherited Flags:\n");
1226 for (FlagParser* p = fp->parent; p; p = p->parent) {
1227 for (size_t i = 0; i < p->flag_count; i++) {
1228 if (!p->flags[i].persistent) continue;
1229 if (is_shadowed_long(fp, p, p->flags[i].name)) continue;
1230 print_flag_row(&p->flags[i], width);
1231 }
1232 }
1233 }
1234
1235 // Subcommands (Immediate children only) ---
1236 if (fp->cmd_count > 0) {
1237 printf("\nAvailable Commands:\n");
1238 size_t cmd_width = calculate_cmd_width(fp);
1239 if (cmd_width < 20) cmd_width = 20; // Minimum width
1240
1241 for (size_t i = 0; i < fp->cmd_count; i++) {
1242 FlagParser* sub = fp->subcommands[i];
1243 printf(" %-*s%s\n", (int)(cmd_width - 2), sub->name,
1244 sub->description ? sub->description : "");
1245 }
1246 }
1247
1248 // --- 4. Footer ---
1249 if (fp->footer) {
1250 printf("\n%s\n", fp->footer);
1251 } else if (fp->cmd_count > 0) {
1252 printf("\nUse \"%s [command] --help\" for more information about a command.\n", fp->name);
1253 }
1254}

1706static void gen_bash_completion(FlagParser* fp, FILE* f) {
1707 if (!fp || !f) return;
1708
1709 char* bin_name = fp->name;
1710
1711 fprintf(f, "#!/usr/bin/env bash\n");
1712 fprintf(f, "# Bash completion for %s\n", bin_name ? bin_name : "program");
1713 fprintf(f, "# Generated by flags.c completion generator\n\n");
1714
1715 fprintf(f, "_");
1716 write_shell_identifier(f, bin_name);
1717 fprintf(f, "_completion() {\n");
1718 fprintf(f, " local cur prev words cword\n");
1719 fprintf(f, " _init_completion || return\n\n");
1720
1721 fprintf(f, " local global_flags=\"");
1722 if (fp->flag_count > 0) {
1723 for (size_t i = 0; i < fp->flag_count; i++) {
1724 fprintf(f, "--");
1725 write_bash_dq(f, fp->flags[i].name);
1726 fprintf(f, " ");
1727 }
1728 }
1729 fprintf(f, "--help\"\n");
1730
1731 // Collect ALL subcommands (recursive) for context detection
1732 fprintf(f, " local subcommands=\"");
1733 write_bash_all_subcommands_list(f, fp);
1734 fprintf(f, "\"\n\n");
1735
1736 // Handle Global Flags Arguments
1737 fprintf(f, " # Handle flags that require arguments\n");
1738 fprintf(f, " case \"$prev\" in\n");
1739 for (size_t i = 0; i < fp->flag_count; i++) {
1740 Flag* flag = &fp->flags[i];
1741 if (flag->type != TYPE_BOOL) {
1742 fprintf(f, " --");
1743 write_bash_pattern(f, flag->name);
1744 if (flag->short_name && isprint(flag->short_name)) {
1745 fprintf(f, "|-");
1746 fputc(flag->short_name, f);
1747 }
1748 fprintf(f, ")\n");
1749
1750 // Type-specific hints
1751 if (flag->type == TYPE_STRING) {
1752 fprintf(f, " COMPREPLY=( $(compgen -f -- \"$cur\") )\n");
1753 }
1754
1755 fprintf(f, " return 0\n");
1756 fprintf(f, " ;;\n");
1757 }
1758 }
1759 fprintf(f, " esac\n\n");
1760
1761 // Context detection - find the active subcommand
1762 fprintf(f, " # Detect active subcommand context\n");
1763 fprintf(f, " local cmd_context=\"\"\n");
1764 fprintf(f, " local i\n");
1765 fprintf(f, " for ((i=1; i < cword; i++)); do\n");
1766 fprintf(f, " local word=\"${words[i]}\"\n");
1767 fprintf(f, " # Skip flags\n");
1768 fprintf(f, " if [[ \"$word\" != -* ]]; then\n");
1769 fprintf(f, " # Check if it's a known subcommand\n");
1770 fprintf(f, " for cmd in $subcommands; do\n");
1771 fprintf(f, " if [[ \"$word\" == \"$cmd\" ]]; then\n");
1772 fprintf(f, " cmd_context=\"$cmd\"\n");
1773 fprintf(f, " break 2\n");
1774 fprintf(f, " fi\n");
1775 fprintf(f, " done\n");
1776 fprintf(f, " fi\n");
1777 fprintf(f, " done\n\n");
1778
1779 // If no subcommand active, show globals + top-level subs
1780 fprintf(f, " if [[ -z \"$cmd_context\" ]]; then\n");
1781 if (fp->cmd_count > 0) {
1782 fprintf(f, " local top_level_subs=\"");
1783 for (size_t i = 0; i < fp->cmd_count; i++) {
1784 write_bash_dq(f, fp->subcommands[i]->name);
1785 fprintf(f, " ");
1786 }
1787 fprintf(f, "\"\n");
1788 fprintf(f,
1789 " COMPREPLY=( $(compgen -W \"$top_level_subs $global_flags\" -- \"$cur\") "
1790 ")\n");
1791 } else {
1792 fprintf(f, " COMPREPLY=( $(compgen -W \"$global_flags\" -- \"$cur\") )\n");
1793 }
1794 fprintf(f, " return 0\n");
1795 fprintf(f, " fi\n\n");
1796
1797 // Subcommand-specific completions
1798 if (fp->cmd_count > 0) {

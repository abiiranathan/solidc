72bool dynarray_push_n(dynarray_t* arr, const void* elements, size_t count) {
73 if (DYNARRAY_UNLIKELY(arr == NULL)) {
74 return false;
75 }
76 if (DYNARRAY_UNLIKELY(count == 0)) {
77 /* Zero-count append is a no-op success even with NULL elements. */
78 return true;
79 }
80 if (DYNARRAY_UNLIKELY(elements == NULL)) {
81 return false;
82 }
83
84 if (DYNARRAY_UNLIKELY(count > SIZE_MAX - arr->size)) {
85 return false;
86 }
87
88 size_t required = arr->size + count;
89
90 if (DYNARRAY_UNLIKELY(required > arr->capacity)) {
91 // Compute 1.5x exponential growth analytically without loops
92 size_t new_capacity = arr->capacity + (arr->capacity >> 1);
93 if (new_capacity < required) {
94 new_capacity = required;
95 }
96
97 if (!dynarray_grow_slowpath(arr, new_capacity)) {
98 return false;
99 }
100 }
101
102 unsigned char* dest = (unsigned char*)arr->data + (arr->size * arr->element_size);
103 memcpy(dest, elements, count * arr->element_size);
104 arr->size = required;
105
106 return true;
107}

155void* dynarray_detach(dynarray_t* arr, size_t* out_size, size_t* out_capacity) {
156 if (out_size) *out_size = 0;
157 if (out_capacity) *out_capacity = 0;
158 if (arr == NULL || arr->data == NULL || arr->size == 0) {
159 return NULL;
160 }
161
162 void* detached = NULL;
163 size_t target_bytes = arr->size * arr->element_size;
164
165 if (arr->capacity == arr->size) {
166 /* Buffer is already exact-size: hand it over directly. */
167 detached = arr->data;
168 arr->data = NULL;
169 } else {
170 /* Shrink to exact size; on realloc failure fall back to a copy so
171 * the caller still receives a valid, owned buffer.
172 * NOTE: a successful realloc consumes the original block -- only
173 * free arr->data when realloc FAILED (the copy branch). */
174 void* shrunk = realloc(arr->data, target_bytes);
175 if (shrunk != NULL) {
176 detached = shrunk;
177 arr->data = NULL;
178 } else {
179 detached = malloc(target_bytes);
180 if (detached == NULL) {
181 /* Nothing lost: the array keeps its buffer. */
182 return NULL;
183 }
184 memcpy(detached, arr->data, target_bytes);
185 free(arr->data);
186 arr->data = NULL;
187 }
188 }
189
190 if (out_size) *out_size = arr->size;
191 if (out_capacity) *out_capacity = arr->size;
192
193 /* Reset to the zero-initialized state: detach consumes the array. */
194 *arr = (dynarray_t){0};
195 return detached;
196}

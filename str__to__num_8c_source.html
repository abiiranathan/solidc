38static inline StoError parse_u64_decimal(const char* p, const char** end_out, uint64_t* out) {
39 uint64_t v = 0;
40 do {
41 unsigned d = (unsigned)(*p - '0');
42 if (v > (UINT64_MAX - d) / 10U) {
43 return STO_OVERFLOW;
44 }
45 v = v * 10U + d;
46 p++;
47 } while (*p >= '0' && *p <= '9');
48 *end_out = p;
49 *out = v;
50 return STO_SUCCESS;
51}

54static inline StoError validate_and_parse_signed(const char* str, int base, intmax_t* result) {
55 if (str == NULL || result == NULL) {
56 return STO_INVALID;
57 }
58
59 const char* p = str;
60 while (is_ascii_space(*p)) {
61 p++;
62 }
63
64 bool neg = false;
65 if (*p == '+') {
66 p++;
67 } else if (*p == '-') {
68 neg = true;
69 p++;
70 }
71
72 if (base == 10 && *p >= '0' && *p <= '9') {
73 /* |INTMAX_MIN| == INTMAX_MAX + 1; computed without signed overflow.
74 * A magnitude above this bound is ERANGE-equivalent (STO_OVERFLOW). */
75 const uint64_t limit = (uint64_t)INTMAX_MAX + (neg ? 1u : 0u);
76 const char* end = NULL;
77 uint64_t mag = 0;
78 StoError err = parse_u64_decimal(p, &end, &mag);
79 if (err != STO_SUCCESS) {
80 return err;
81 }
82 /* libc checks range before consuming-trailing-garbage validity:
83 * "99999999999999999999x" is ERANGE, not INVALID. Match that. */
84 if (mag > limit) {
85 return STO_OVERFLOW;
86 }
87 if (*end != '\0') {
88 return STO_INVALID;
89 }
90 *result = neg ? (intmax_t)(UINTMAX_MAX - mag + 1) : (intmax_t)mag;
91 return STO_SUCCESS;
92 }
93
94 char* endptr = NULL;
95 errno = 0;
96 *result = strtoimax(str, &endptr, base);
97
98 // Check for standard ERANGE (overflow of intmax_t)
99 if (errno == ERANGE) {
100 return STO_OVERFLOW;
101 }
102
103 // Check for invalid input (no conversion or partial conversion)
104 if (endptr == str || *endptr != '\0') {
105 return STO_INVALID;
106 }
107
108 return STO_SUCCESS;
109}

112static inline StoError validate_and_parse_unsigned(const char* str, int base, uintmax_t* result) {
113 if (str == NULL || result == NULL) {
114 return STO_INVALID;
115 }
116
117 // Skip whitespace to correctly find the sign
118 const char* p = str;
119 while (is_ascii_space(*p)) {
120 p++;
121 }
122
123 // Check for negative input manually.
124 // Standard strtoumax wraps negative numbers (e.g. "-1" -> UINTMAX_MAX).
125 // For a strict "string to unsigned" conversion, negative input is an underflow.
126 if (*p == '-') {
127 return STO_UNDERFLOW;
128 }
129
130 if (base == 10 && *p >= '0' && *p <= '9') {
131 const char* end = NULL;
132 uint64_t v = 0;
133 StoError err = parse_u64_decimal(p, &end, &v);
134 if (err != STO_SUCCESS) {
135 return err;
136 }
137 if (*end != '\0') {
138 return STO_INVALID;
139 }
140 *result = (uintmax_t)v;
141 return STO_SUCCESS;
142 }
143
144 char* endptr = NULL;
145 errno = 0;
146 *result = strtoumax(str, &endptr, base);
147
148 // Check for standard conversion errors (overflow of uintmax_t)
149 if (errno == ERANGE) {
150 return STO_OVERFLOW;
151 }
152
153 // Check for invalid input (no conversion or partial conversion)
154 if (endptr == str || *endptr != '\0') {
155 return STO_INVALID;
156 }
157
158 return STO_SUCCESS;
159}

300StoError str_to_float(const char* str, float* result) {
301 if (str == NULL || result == NULL) {
302 return STO_INVALID;
303 }
304 char* endptr = NULL;
305 errno = 0;
306 *result = strtof(str, &endptr);
307 if (endptr == str || *endptr != '\0') return STO_INVALID;
308 if (errno == ERANGE) {
309 if (isinf(*result) || fabsf(*result) > FLT_MAX) return STO_OVERFLOW;
310 if (*result == 0.0f || fabsf(*result) < FLT_MIN) return STO_UNDERFLOW;
311 }
312
313 return STO_SUCCESS;
314}

331StoError str_to_bool(const char* str, bool* result) {
332 if (str == NULL || result == NULL) {
333 return STO_INVALID;
334 }
335
336 /*
337 * First-character dispatch: at most one candidate per starting letter,
338 * so a single lowercase-and-switch replaces up to eight strcasecmp
339 * calls. "1"/"0" must match the WHOLE string (no trailing garbage).
340 */
341 switch ((char)(str[0] | 0x20)) {
342 case 't': {
343 if (strcasecmp(str, "true") == 0) {
344 *result = true;
345 return STO_SUCCESS;
346 }
347 break;
348 }
349 case 'f': {
350 if (strcasecmp(str, "false") == 0) {
351 *result = false;
352 return STO_SUCCESS;
353 }
354 break;
355 }
356 case 'y': {
357 if (strcasecmp(str, "yes") == 0) {
358 *result = true;
359 return STO_SUCCESS;
360 }
361 break;
362 }
363 case 'n': {
364 if (strcasecmp(str, "no") == 0) {
365 *result = false;
366 return STO_SUCCESS;
367 }
368 break;
369 }
370 case 'o': {
371 if (strcasecmp(str, "on") == 0) {
372 *result = true;
373 return STO_SUCCESS;
374 }
375 if (strcasecmp(str, "off") == 0) {
376 *result = false;
377 return STO_SUCCESS;
378 }
379 break;
380 }
381 case '1': {
382 if (str[1] == '\0') {
383 *result = true;
384 return STO_SUCCESS;
385 }
386 break;
387 }
388 case '0': {
389 if (str[1] == '\0') {
390 *result = false;
391 return STO_SUCCESS;
392 }
393 break;
394 }
395 default:
396 break;
397 }
398
399 return STO_INVALID;
400}

256static ARENA_INLINE void arena_reset(Arena* a) {
257 if (!a || !a->head) return;
258
259 /* Dirty check: if we're still on the head block and the cursor hasn't
260 * moved off its base, no allocation has happened since the last reset.
261 * Skip the writes entirely rather than re-storing identical values.
262 * head_base is cached on the hot path so we avoid a second cache-line
263 * touch through the head block when the arena is already clean. */
264 if (a->current_block == a->head && a->curr == a->head_base) return;
265
266 a->current_block = a->head;
267 a->curr = a->head_base;
268 a->end = a->head->end;
269}

351static ARENA_INLINE size_t arena_used_size(const Arena* a) {
352 if (!a || !a->current_block) return 0;
353 size_t block_capacity = (size_t)(a->current_block->end - a->current_block->base);
354 size_t block_used = (size_t)(a->curr - a->current_block->base);
355 return (a->total_committed - block_capacity) + block_used;

396static ARENA_NONNULL_ ARENA_INLINE ARENA_ATTR_MALLOC
397ARENA_ATTR_ALLOC_SIZE(2) void* arena_alloc_align(Arena* arena, size_t size, size_t alignment) {
398#ifdef ARENA_ABORT_ON_OOM
399 /* Preserve the returns_nonnull contract: treat size==0 as size==1. */
400 if (size == 0) size = 1;
401#else
402 if (size == 0) return NULL;
403#endif
404
405 uintptr_t curr = (uintptr_t)arena->curr;
406 uintptr_t aligned = (curr + alignment - 1) & ~(uintptr_t)(alignment - 1);
407 uintptr_t next = aligned + size;
408
409 if (ARENA_LIKELY(next <= (uintptr_t)arena->end)) {
410 arena->curr = (char*)next;
411 /* Prefetch the allocated region; hides DRAM latency when the caller
412 * immediately writes the object (the common case). */
413 ARENA_PREFETCH((void*)aligned);
414 return (void*)aligned;
415 }
416
417 void* ptr = _arena_alloc_slow(arena, size, alignment);
418 if (ARENA_UNLIKELY(!ptr)) {
419#ifdef ARENA_ABORT_ON_OOM
420 ARENA_OOM_HANDLER(size);
421 ARENA_UNREACHABLE();
422#else
423 return NULL;
424#endif
425 }
426 return ptr;
427}

455static ARENA_NONNULL_ ARENA_INLINE ARENA_ATTR_MALLOC
456ARENA_ATTR_ALLOC_SIZE(2) void* arena_alloc_unaligned(Arena* arena, size_t size) {
457#ifdef ARENA_ABORT_ON_OOM
458 if (size == 0) size = 1;
459#else
460 if (size == 0) return NULL;
461#endif

463 char* ptr = arena->curr;
464 char* next = ptr + size;

467 arena->curr = next;
468 ARENA_PREFETCH(ptr);

511static ARENA_INLINE void* arena_realloc(Arena* arena, void* old_ptr, size_t old_size,
512 size_t new_size, size_t alignment) {
513 if (!old_ptr || old_size == 0) {
514 return arena_alloc_align(arena, new_size, alignment);
515 }
516
517 if (new_size == 0) {
518 /* Shrink to nothing: reclaim bytes only if this was the last alloc. */
519 if ((char*)old_ptr + old_size == arena->curr) {
520 arena->curr = (char*)old_ptr;
521 }
522 /* NULL here is a defined contract value, not an OOM — do not invoke
523 * ARENA_OOM_HANDLER. */
524 return NULL;
525 }
526
527 /* Fast path: old_ptr was the last allocation in the current block. */
528 if ((char*)old_ptr + old_size == arena->curr) {
529 char* new_end = (char*)old_ptr + new_size;
530 if (new_size <= old_size || new_end <= arena->end) {
531 arena->curr = new_end;
532 return old_ptr;
533 }
534 }
535
536 /* Slow path: fresh allocation + copy; old bytes are stranded until reset. */
537 void* new_ptr = arena_alloc_align(arena, new_size, alignment);
538 if (ARENA_UNLIKELY(!new_ptr)) {
539 /* arena_alloc_align already invoked ARENA_OOM_HANDLER if the macro is
540 * set, so we only reach here in the non-abort build. */
541 return NULL;
542 }
543 memcpy(new_ptr, old_ptr, old_size < new_size ? old_size : new_size);
544 return new_ptr;
545}

623static ARENA_INLINE bool arena_alloc_batch(Arena* arena, const size_t* sizes, size_t count,
624 void** out_ptrs) {
625 if (!arena || !sizes || !out_ptrs || count == 0) return false;
626
627 const size_t mask = ARENA_DEFAULT_ALIGN - 1;
628 size_t total = 0;
629 for (size_t i = 0; i < count; ++i) {
630 size_t rounded = (total + mask) & ~mask;
631 /* Detect integer overflow in the size accumulation. This is a
632 * caller bug, not OOM — return false without invoking the OOM handler
633 * regardless of ARENA_ABORT_ON_OOM. */
634 if (ARENA_UNLIKELY(sizes[i] > (size_t)-1 - rounded)) return false;
635 total = rounded + sizes[i];
636 }
637
638 char* base = (char*)arena_alloc(arena, total);
639 if (ARENA_UNLIKELY(!base)) {
640 /* arena_alloc already called ARENA_OOM_HANDLER if the macro is set. */
641 return false;
642 }
643
644 char* cur = base;
645 for (size_t i = 0; i < count; ++i) {
646 cur = (char*)(((uintptr_t)cur + mask) & ~mask);
647 out_ptrs[i] = cur;
648 cur += sizes[i];
649 }
650 return true;
651}

669static ARENA_INLINE char* arena_strdup(Arena* arena, const char* str) {
670 /* NULL arguments are a programming error, not OOM. Return NULL silently
671 * (or rely on the caller to guard); do not invoke ARENA_OOM_HANDLER. */
672 if (ARENA_UNLIKELY(!arena || !str)) return NULL;
673 size_t len = strlen(str);
674 char* dup = (char*)arena_alloc_unaligned(arena, len + 1);
675 if (ARENA_UNLIKELY(!dup)) return NULL;
676 memcpy(dup, str, len + 1);
677 return dup;
678}

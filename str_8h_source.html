54static inline const char* str_search_impl(const char* hs, size_t hlen, const char* nd, size_t nlen) {
55 if (nlen == 0) return hs;
56 if (hlen < nlen) return NULL;
57
58 /* Optimized fast path for single-character search using vectorized system memchr */
59 if (nlen == 1) return (const char*)memchr(hs, (unsigned char)nd[0], hlen);
60
61 const char* cur = hs;
62 const char* end = hs + hlen - nlen;
63 unsigned char first = (unsigned char)nd[0];
64 unsigned char last = (unsigned char)nd[nlen - 1];
65
66 /* Slotted window search: scan for first byte, verify last byte, then compare inner bytes */
67 while (cur <= end) {
68 /* Rapidly skip non-matching characters using memchr */
69 cur = (const char*)memchr(cur, first, (size_t)(end - cur + 1));
70 if (!cur) return NULL;
71
72 /* Filter false positives early by testing the last byte of needle */
73 if ((unsigned char)cur[nlen - 1] == last) {
74 if (nlen <= 9) {
75 /* Unrolled manual loop for short needles to avoid call overhead of memcmp */
76 size_t i = 0;
77 for (; i < nlen - 2; i++) {
78 if (cur[1 + i] != nd[1 + i]) goto next_iter;
79 }
80 return cur;
81 } else {
82 /* Bulk comparison for larger needle payloads */
83 if (memcmp(cur + 1, nd + 1, nlen - 2) == 0) return cur;
84 }
85 }
86 next_iter:
87 cur++;
88 }
89 return NULL;
90}

1078 return str_dup(str);
1079 }
1080
1081 /* Step 2: Compute target memory bounds */
1082 size_t result_len;
1083 if (new_len >= old_len) {
1084 size_t diff = new_len - old_len;
1085 if (diff > 0 && count > (SIZE_MAX - hlen) / diff) goto oom;
1086 result_len = hlen + count * diff;
1087 } else {
1088 result_len = hlen - count * (old_len - new_len);
1089 }
1090
1091 char* r = (char*)malloc(result_len + 1);
1092 if (!r) goto oom;
1093
1094 /* Step 3: Stitch together target string in single copy pass */
1095 size_t wp = 0, sp = 0;
1096 for (size_t i = 0; i < count; i++) {
1097 size_t gap = offs[i] - sp;
1098 if (gap) {
1099 memcpy(r + wp, str + sp, gap);
1100 wp += gap;
1101 }
1102 if (new_len) {
1103 memcpy(r + wp, new_sub, new_len);
1104 wp += new_len;
1105 }
1106 sp = offs[i] + old_len;
1107 }
1108 size_t tail = hlen - sp;
1109 if (tail) {
1110 memcpy(r + wp, str + sp, tail);
1111 wp += tail;
1112 }
1113 r[wp] = '\0';
1114
1115 if (offs != stack_offs) free(offs);
1116 return r;
1117
1118oom:
1119 if (offs != stack_offs) free(offs);
1120 return NULL;

1122}

1133static inline char** str_split(const char* str, const char* delim, size_t* count_out) {
1134 if (!count_out) return NULL;
1135 *count_out = 0;
1136 if (!str) return NULL;
1137
1138 /* Empty or NULL delimiter returns single element array copy of entire input */
1139 if (!delim || !*delim) {
1140 char** r = (char**)malloc(2 * sizeof(char*));
1141 if (!r) return NULL;
1142 r[0] = str_dup(str);

1155 if (!result) return NULL;

1158 size_t rem = strlen(str);
1159 size_t count = 0;
1160
1161 if (dlen == 1) {
1162 /* Single-character SIMD fast path */
1163 unsigned char target = (unsigned char)delim[0];
1164 for (;;) {
1165 const char* match = (const char*)memchr(start, target, rem);
1166 size_t tok_len = match ? (size_t)(match - start) : rem;
1167
1168 if (count + 1 >= cap) {
1169 if (cap > SIZE_MAX / 2 / sizeof(char*)) goto split_err;
1170 size_t new_cap = cap * 2;
1171 char** tmp = (char**)realloc(result, new_cap * sizeof(char*));
1172 if (!tmp) goto split_err;
1173 result = tmp;
1174 cap = new_cap;
1175 }
1176
1177 result[count] = (char*)malloc(tok_len + 1);
1178 if (!result[count]) goto split_err;
1179 memcpy(result[count], start, tok_len);
1180 result[count][tok_len] = '\0';
1181 count++;
1182
1183 if (!match) break;
1184 start = match + 1;
1185 rem -= (tok_len + 1);
1186 }
1187 } else {
1188 /* Multi-character substring search split path */
1189 const char* end = str + rem;
1190 for (;;) {
1191 const char* found = str_search_impl(start, (size_t)(end - start), delim, dlen);
1192 const char* tok_end = found ? found : end;
1193
1194 if (count + 1 >= cap) {
1195 if (cap > SIZE_MAX / 2 / sizeof(char*)) goto split_err;
1196 size_t new_cap = cap * 2;
1197 char** tmp = (char**)realloc(result, new_cap * sizeof(char*));
1198 if (!tmp) goto split_err;
1199 result = tmp;
1200 cap = new_cap;
1201 }
1202
1203 size_t tok_len = (size_t)(tok_end - start);
1204 result[count] = (char*)malloc(tok_len + 1);
1205 if (!result[count]) goto split_err;
1206 memcpy(result[count], start, tok_len);
1207 result[count][tok_len] = '\0';
1208 count++;
1209
1210 if (!found) break;
1211 start = found + dlen;
1212 }
1213 }
1214
1215 result[count] = NULL; /* Append mandatory NULL sentinel pointer */
1216 *count_out = count;
1217 return result;
1218
1219split_err:
1220 for (size_t i = 0; i < count; i++) free(result[i]);
1221 free(result);
1222 return NULL;
1223}

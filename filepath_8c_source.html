24// Generate a random string for temporary file/directory names.
25// len must be < 64 bytes.
26/*
27 * Thread-safety notes: entropy is provided by fp_random_bytes() which is
28 * implemented per platform. On Linux it is getrandom(2), which is
29 * thread-safe and requires no shared state, so no locking is done at all —
30 * the earlier version held a mutex across the syscall (pure contention) and
31 * initialised it with a check-then-act pattern that could double-initialise
32 * under concurrency (Bug #9). Other platforms keep their own locking inside
33 * the backend.
34 */
37void fp_random_name(char* str, size_t len) {
38 static const char charset[] = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
39 const size_t charset_size = sizeof(charset) - 1; // Exclude null terminator
40
41 unsigned char buffer[64]; // Buffer for random bytes (sufficient for typical len)
42
43 // Ensure we don't write beyond requested length
44 size_t bytes_needed = len < 64 ? len : 64;
45
46 if (!fp_random_bytes(buffer, bytes_needed)) {
47 str[0] = '\0'; // Fallback to empty string on error
48 return;
49 }
50
51 // Convert random bytes to characters from charset
52 for (size_t i = 0; i < len; i++) {
53 str[i] = charset[buffer[i % bytes_needed] % charset_size];
54 }
55 str[len] = '\0';
56}

95bool filepath_makedirs(const char* path) {
96 if (!path || *path == '\0') {
97 errno = EINVAL;
98 return false;
99 }
100
101 char* temp_path = strdup(path);
102 if (!temp_path) {
103 errno = ENOMEM;
104 return false;
105 }
106
107 char* p = temp_path;
108 while (*p != '\0') {
109 while (*p == '/' || *p == '\\') {
110 p++;
111 }
112 while (*p != '\0' && *p != '/' && *p != '\\') {
113 p++;
114 }
115
116 char old = *p;
117 *p = '\0';
118
119 if (*temp_path != '\0') {
120 if (dir_create(temp_path) != 0) {
121 free(temp_path);
122 return false;
123 }
124 }
125
126 *p = old;
127 if (*p != '\0') {
128 p++;
129 }
130 }
131
132 free(temp_path);
133 return true;
134}

230// List files in a directory with unified error handling
233char** dir_list(const char* path, size_t* count) {
234 if (!path || !count || *path == '\0') {
235 errno = EINVAL;
236 return NULL;
237 }
238
239 Directory* dir = NULL;
240 char** list = NULL;
241 size_t size = 0;
242 size_t capacity = 10;
243 char* name = NULL;
244
245 dir = dir_open(path);
246 if (!dir) return NULL;
247
248 list = (char**)calloc(capacity, sizeof(char*));
249 if (!list) {
250 goto error;
251 }
252
253 while ((name = dir_next(dir)) != NULL) {
254 if (size >= capacity) {
255 capacity *= 2;
256 char** tmp = (char**)realloc(list, capacity * sizeof(char*));
257 if (!tmp) {
258 goto error;
259 }
260 list = tmp;
261 }
262
263 list[size] = strdup(name);
264 if (!list[size]) {
265 goto error;
266 }
267 size++;
268 }
269
270 dir_close(dir);
271 *count = size;
272 return list;
273
274error:
275 if (list) {
276 // Free all successfully duplicated names
277 for (size_t i = 0; i < size; i++) {
278 free(list[i]);
279 }
280 free(list);
281 }
282
283 if (dir) dir_close(dir);
284
285 return NULL;
286}

389char* filepath_expanduser(const char* path) {
390 if (!path) {
391 errno = EINVAL;
392 return NULL;
393 }
394
395 if (path[0] != '~') {
396 return strdup(path);
397 }
398
399 const char* home = user_home_dir();
400 if (!home) {
401 errno = ENOENT;
402 return NULL;
403 }
404
405 size_t pathLen = strlen(path);
406 bool isHome = pathLen == 1 || (pathLen == 2 && (path[1] == '/' || path[1] == '\\'));
407 if (isHome) {
408 return strdup(home);
409 }
410
411 size_t len = strlen(home) + pathLen + 1;
412 char* expanded = (char*)malloc(len);
413 if (!expanded) {
414 errno = ENOMEM;
415 return NULL;
416 }
417
418 const char* suffix = path + 1;
419 // Skip leading separator after ~
420 if (*suffix == '/' || *suffix == '\\') {
421 suffix++;
422 }
423 snprintf(expanded, len, "%s%c%s", home, PATH_SEP, suffix);
424 return expanded;
425}

431bool filepath_expanduser_buf(const char* path, char* expanded, size_t len) {

450 if (isHome) {

112static DYNARRAY_INLINE bool dynarray_push(dynarray_t* arr, const void* element) {
113 if (DYNARRAY_UNLIKELY(arr == NULL || element == NULL)) {
114 return false;
115 }
116
117 if (DYNARRAY_UNLIKELY(arr->size >= arr->capacity)) {
118 // Fast 1.5x capacity growth via bitwise shifts: cap + (cap >> 1)
119 size_t new_cap = arr->capacity + (arr->capacity >> 1);
120 if (DYNARRAY_UNLIKELY(new_cap <= arr->capacity)) {
121 new_cap = arr->capacity + 1;
122 }
123 if (DYNARRAY_UNLIKELY(!dynarray_grow_slowpath(arr, new_cap))) {
124 return false;
125 }
126 }
127
128 unsigned char* dest = (unsigned char*)arr->data + (arr->size * arr->element_size);
129 dynarray_fast_copy(dest, element, arr->element_size);
130 arr->size++;
131
132 return true;
133}

143static DYNARRAY_INLINE bool dynarray_pop(dynarray_t* arr, void* out_element) {
144 if (DYNARRAY_UNLIKELY(arr == NULL || arr->size == 0)) {
145 return false;
146 }
147
148 arr->size--;
149
150 if (out_element != NULL) {
151 const unsigned char* src = (const unsigned char*)arr->data + (arr->size * arr->element_size);
152 dynarray_fast_copy(out_element, src, arr->element_size);
153 }
154
155 // Shrink check on cold path if usage drops below threshold
156 if (DYNARRAY_UNLIKELY(arr->capacity > DYNARRAY_INITIAL_CAPACITY &&
157 arr->size < arr->capacity / DYNARRAY_SHRINK_THRESHOLD)) {
158 size_t new_capacity = arr->capacity >> 1; // Faster capacity / 2
159 if (new_capacity < DYNARRAY_INITIAL_CAPACITY) {
160 new_capacity = DYNARRAY_INITIAL_CAPACITY;
161 }
162 dynarray_grow_slowpath(arr, new_capacity);
163 }
164
165 return true;
166}

241static DYNARRAY_INLINE bool dynarray_remove(dynarray_t* arr, size_t index) {
242 if (DYNARRAY_UNLIKELY(arr == NULL || index >= arr->size)) {
243 return false;
244 }
245
246 unsigned char* base = (unsigned char*)arr->data;
247 memmove(base + (index * arr->element_size), base + ((index + 1) * arr->element_size),
248 (arr->size - index - 1) * arr->element_size);

270static DYNARRAY_INLINE bool dynarray_swap_remove(dynarray_t* arr, size_t index, void* out_element) {
271 if (DYNARRAY_UNLIKELY(arr == NULL || index >= arr->size)) {
272 return false;
273 }
274
275 unsigned char* base = (unsigned char*)arr->data;
276 unsigned char* slot = base + (index * arr->element_size);

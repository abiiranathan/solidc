49bool fp_random_bytes(unsigned char* buf, size_t n) {
50 HCRYPTPROV hCryptProv;
51 if (!CryptAcquireContextW(&hCryptProv, NULL, NULL, PROV_RSA_FULL, CRYPT_VERIFYCONTEXT)) {
52 return false;
53 }
54
55 if (!CryptGenRandom(hCryptProv, (DWORD)n, buf)) {
56 CryptReleaseContext(hCryptProv, 0);
57 return false;
58 }
59 CryptReleaseContext(hCryptProv, 0);
60 return true;
61}

67// Open a directory
71Directory* dir_open(const char* path) {
72 if (!path || *path == '\0') {
73 errno = EINVAL;
74 return NULL;
75 }
76
77 Directory* dir = (Directory*)calloc(1, sizeof(Directory));
78 if (!dir) {
79 errno = ENOMEM;
80 return NULL;
81 }
82
83 dir->path = strdup(path);
84 if (!dir->path) {
85 free(dir);
86 errno = ENOMEM;
87 return NULL;
88 }
89
90 wchar_t wpath[MAX_PATH];
91 if (!MultiByteToWideChar(CP_UTF8, 0, path, -1, wpath, MAX_PATH)) {
92 free(dir->path);
93 free(dir);
94 errno = EINVAL; // More specific error code
95 return NULL;
96 }
97
98 wchar_t search_path[MAX_PATH + 2];
99 if (swprintf(search_path, MAX_PATH + 2, L"%ls\\*", wpath) < 0) {
100 free(dir->path);
101 free(dir);
102 errno = ENAMETOOLONG;
103 return NULL;
104 }
105
106 dir->handle = FindFirstFileW(search_path, &dir->find_data);
107 if (dir->handle == INVALID_HANDLE_VALUE) {
108 free(dir->path);
109 free(dir);
110 errno = GetLastError() == ERROR_FILE_NOT_FOUND ? ENOENT : EACCES;
111 return NULL;
112 }
113 return dir;
114}

157static void map_win32_attrs(const WIN32_FIND_DATAW* fd, FileAttributes* attr) {
158 attr->attrs = FATTR_NONE;
159 if (fd->dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY)
160 attr->attrs |= FATTR_DIR;
161 else
162 attr->attrs |= FATTR_FILE;
163 if (fd->dwFileAttributes & FILE_ATTRIBUTE_REPARSE_POINT) attr->attrs |= FATTR_SYMLINK;
164 if (fd->dwFileAttributes & FILE_ATTRIBUTE_HIDDEN) attr->attrs |= FATTR_HIDDEN;
165 if (fd->cFileName[0] == '.') attr->attrs |= FATTR_HIDDEN;
166
167 attr->size = ((size_t)fd->nFileSizeHigh << 32) | fd->nFileSizeLow;
168
169 // Convert Windows FileTime to Unix mtime (simplified)
170 ULARGE_INTEGER ull;
171 ull.LowPart = fd->ftLastWriteTime.dwLowDateTime;
172 ull.HighPart = fd->ftLastWriteTime.dwHighDateTime;
173 attr->mtime = (time_t)((ull.QuadPart - 116444736000000000ULL) / 10000000ULL);
174}

433static int dir_walk_win32(const char* path, WalkDirCallback callback, void* data, int depth) {
434 if (depth > FP_MAX_DIR_DEPTH) {
435 errno = ELOOP;
436 return -1;
437 }
438
439 Directory* dir = dir_open(path);
440 if (!dir) return -1;
441
442 char fullpath[FILENAME_MAX];
443 int status = 0;
444
445 do {
446 const wchar_t* wname = dir->find_data.cFileName;
447 if (wcscmp(wname, L".") == 0 || wcscmp(wname, L"..") == 0) continue;
448
449 char name[MAX_PATH];
450 WideCharToMultiByte(CP_UTF8, 0, wname, -1, name, MAX_PATH, NULL, NULL);
451
452 if (!filepath_join_buf(path, name, fullpath, sizeof(fullpath))) {
453 status = -1;
454 break;
455 }
456
457 FileAttributes attr;
458 map_win32_attrs(&dir->find_data, &attr);
459
460 WalkDirOption opt = callback(&attr, fullpath, name, data);
461 if (opt == DirStop) break;
462 if (opt == DirError) {
463 status = -1;
464 break;
465 }
466 if (opt == DirSkip) continue;
467
468 if (fattr_is_dir(&attr)) {
469 if (dir_walk_win32(fullpath, callback, data, depth + 1) != 0) {
470 status = -1;
471 break;
472 }
473 }
474 } while (FindNextFileW(dir->handle, &dir->find_data));
475
476 dir_close(dir);
477 return status;
478}

5/* ============================================================================
6 * threadpool.c — Work-Stealing Threadpool
7 * ============================================================================
8 *
9 * ARCHITECTURE
10 * ------------
11 *
12 * submitter(s) N workers, each owning a private deque
13 * │ ┌────────────────────────────────────┐
14 * │ threadpool_submit(_batch)│ W0 W1 W2 ... Wn-1 │
15 * ▼ │ ▲ ▲ ▲ ▲ │
16 * ┌──────────────────┐ batch │ │ │ │ steal │ │
17 * │ GlobalQueue gq │──pull(64)──▶│ │ │ │ (CAS) │ │
18 * │ mutex + ring │ │ ▼ ▼ ▼ ▼ │
19 * └──────────────────┘ │ LIFO pop / FIFO steal deques │
20 * ▲ └────────────────────────────────────┘
21 * └──────── unpark_n / cascade wakeups ◀───── worker_park on idle
22 *
23 * Every worker owns a Chase-Lev work-stealing deque:
24 *
25 * OWNER — pushes and pops at BOTTOM. No lock; a release store publishes.
26 * THIEVES— steal from TOP via a seq_cst CAS. Losers retry elsewhere.
27 *
28 * External threads push into GlobalQueue, a mutex-protected MPMC ring.
29 * Workers drain it in batches of BATCH_SIZE: the first task executes
30 * immediately, the rest are pushed onto the thief's own deque so the cost
31 * of one mutex acquisition amortises over up to BATCH_SIZE tasks.
32 *
33 * MEMORY LAYOUT / FALSE SHARING
34 * -----------------------------
35 * Every hot field sits on its own cache line (CACHE_ALIGNED):
36 * - WorkStealDeque.bottom (owner-written only) and .top (CAS by thieves)
37 * live on separate lines so owner pops never bounce against steals.
38 * - Each worker struct is line-aligned; workers[] entries are pointers,
39 * so scanning victims touches only their deque headers.
40 * - park_lock / num_parked / idle_lock / num_pending each get their own
41 * line inside Threadpool.
42 *
43 * THE OWNERSHIP INVARIANT — num_pending
44 * -------------------------------------
45 * A single atomic counter, num_pending, counts tasks that have been
46 * accepted but not yet completed. It is the SOLE source of truth for
47 * idleness ("num_pending == 0") used by threadpool_wait(), destroy() and
48 * worker parking. Two rules make that sound:
49 *
50 * 1. Credit BEFORE enqueue. Submitters increment num_pending before the
51 * task becomes visible to any consumer (and debit for tasks rejected
52 * due to shutdown/overflow). A task therefore never exists without
53 * outstanding credit, so pending can never hit zero early. This also
54 * makes the accounting wrap-proof: a completion decrement can never
55 * race below zero on a task that has not been credited yet.
56 *
57 * 2. Debit at completion, in batches. Workers accumulate completions in
58 * local_done and flush to num_pending every PENDING_FLUSH_BATCH tasks
59 * or at drain points (no work found / before parking / exit). Pending
60 * may be OVERSTATED between flushes (waiters just wait slightly
61 * longer); it can never be understated, which is what correctness
62 * requires. Batching keeps the shared line off the per-task path.
63 *
64 * This design eliminated an entire class of false-idle races that plagued
65 * earlier revisions (Bug #4/Bug #5): those scanned gq head/tail and every
66 * deque's bottom/top to decide idleness, and transient states mid-pop /
67 * mid-steal / mid-batch-distribution made "queues look empty while a task
68 * is in flight" observable. With num_pending, no queue inspection is
69 * needed at all.
70 *
71 * PARKING & THE WAKEUP PROTOCOL
72 * -----------------------------
73 * Idle workers first run an exponential PAUSE backoff (YIELD_THRESHOLD
74 * rounds), then park on Condition work_available under park_lock.
75 *
76 * Bug #6 (lost wakeup): a racing submitter and a parking worker can both
77 * observe stale zeros on two DIFFERENT variables (pending vs parked)
78 * unless their four ops share a total order. The fix marks exactly four
79 * operations seq_cst — worker_park's parked++, its pending check, the
80 * submitter's pending++, and unpark_*'s parked read. If the worker's
81 * check reads pre-increment pending, seq_cst order forces the submitter's
82 * later parked read to see >0, so a signal is guaranteed. On x86 this is
83 * free: seq_cst loads are plain MOV and the RMW was already LOCK'd.
84 *
85 * Wakeup volume scales with work size:
86 * - threadpool_submit(): wakes ONE worker.
87 * - threadpool_submit_batch(): wakes count/BATCH_SIZE + 1 workers (Perf
88 * #7a) — waking a single worker let it drain entire multi-thousand-task
89 * submissions alone at single-thread speed.
90 * - try_steal() after pulling a global-queue batch wakes one more worker
91 * (cascade propagation), giving O(work) fan-out with zero extra
92 * submission-side signalling.
93 *
94 * Bug #7 (producer deadlock): when the global queue fills, submitters
95 * sleep on Condition not_full — but consumers were only woken after the
96 * whole submission finished streaming. A batch larger than GLOBAL_Q_SIZE
97 * deadlocked whenever all workers had already parked. Producers now call
98 * unpark_all() immediately before cond_wait(not_full).
99 *
100 * IDLE DETECTION FOR WAITERS
101 * --------------------------
102 * Completing workers broadcast all_idle under idle_lock when their flush
103 * drives num_pending to 0. threadpool_wait() first checks the fast path,
104 * then adaptively spins ~ADAPTIVE_SPIN_NS polling pending (Perf #8 — in
105 * ping-pong workloads >60% of wall time was futex sys-time), then falls
106 * back to cond_wait. The broadcast-under-idle_lock handshake makes the
107 * slow path lossless: a completing worker cannot emit the final broadcast
108 * between the waiter's predicate check and its sleep.
109 *
110 * BACKOFF POLICY (Perf #9)
111 * ------------------------
112 * strace profiling showed the previous middle tier of sched_yield() calls
113 * cost ~16 yield syscalls (~45us service time each) per submitted task —
114 * losing threads paid kernel timeslices to retry almost immediately. The
115 * backoff is now pure PAUSE loops (2..256 pauses) followed directly by a
116 * proper park; measured ping-pong latency improved 13x at 1 worker.
117 *
118 * BUG HISTORY (abridged; see inline comments)
119 * -------------------------------------------
120 * #1 external+worker concurrent deque_push_bottom corruption -> GlobalQueue
121 * #2 stale TLS worker index across pool lifetimes -> SIZE_MAX guard
122 * #3 steal before workers[] fully populated -> startup barrier
123 * #4/#5 false-idle races orphaning tasks -> num_pending redesign
124 * #6 lost wakeup across two variables -> seq_cst quartet
125 * #7 producer deadlock on full queue -> unpark before sleep
126 *
127 * TUNABLES
128 * --------
129 * DEQUE_SIZE private deque capacity (power of 2). 16384.
130 * GLOBAL_Q_SIZE shared queue capacity (power of 2). 65536.
131 * Note: submissions larger than this stream through
132 * in chunks and block on not_full when full.
133 * BATCH_SIZE tasks pulled from the global queue per lock hold.
134 * PENDING_FLUSH_BATCH completions accumulated before flushing the
135 * shared pending counter.
136 * YIELD_THRESHOLD PAUSE-backoff rounds before parking.
137 * ADAPTIVE_SPIN_NS bounded poll duration in threadpool_wait().
138 *
139 * BENCHMARK EVIDENCE (see benchmarks/threadpool_micro.c)
140 * ------------------------------------------------------
141 * pingpong (submit 1 task + wait): 4034ns -> 304ns @1T (13x)
142 * burst drain tail (65536 no-op tasks): deadlock-free, ~0.9ms @8T
143 * mproducers (4x262144 tiny tasks): 33ns/task @2T -> 20ns/task @8T
144 * granular sweep: 4.6x @2T .. 6.7x @8T for >=220ns
145 * tasks; <25ns tasks are producer-bound
146 * ============================================================================ */

203/* ── Chase-Lev private deque ─────────────────────────────────────────────── */
204/*
205 * Single-producer (owner), multi-consumer LIFO/FIFO deque.
206 *
207 * bottom: written ONLY by the owner (relaxed reads, release writes).
208 * top: read by everyone; advanced by owner (last-item pop) and thieves
209 * via seq_cst CAS — the linearisation point of every removal.
210 *
211 * Indices are unbounded size_t; only b - t is meaningful, so wraparound at
212 * 2^64 is harmless. Slots are addressed with & DEQUE_MASK.
213 */
214typedef struct {
215 CACHE_ALIGNED atomic_size_t bottom;
216 CACHE_ALIGNED atomic_size_t top;
217 CACHE_ALIGNED Task tasks[DEQUE_SIZE];
218} WorkStealDeque;
219
220/* ── Global submission queue ─────────────────────────────────────────────── */
221/*
222 * Mutex-protected MPMC ring for external submitters (and overflow from
223 * worker-side batch submission). head/tail are uint32_t counters stored
224 * MASKED; all arithmetic is done on (h - t) & GLOBAL_Q_MASK so wraparound
225 * is safe. One slot of capacity is sacrificed as the full/empty sentinel.
226 *
227 * not_full: producers blocked because the ring filled. Consumers broadcast
228 * after every successful pull; producers also unpark consumers before
229 * sleeping here (Bug #7).
230 */
231typedef struct {
232 CACHE_ALIGNED Lock mutex;
233 CACHE_ALIGNED atomic_uint head;
234 CACHE_ALIGNED atomic_uint tail;
235 CACHE_ALIGNED Task tasks[GLOBAL_Q_SIZE];
236 Condition not_full;
237 struct Threadpool* pool;
238} GlobalQueue;

240/* ── Per-worker ──────────────────────────────────────────────────────────── */
241typedef struct worker {
242 CACHE_ALIGNED WorkStealDeque deque;
243 CACHE_ALIGNED Thread pthread;
244 CACHE_ALIGNED size_t index;
245 struct Threadpool* pool;
246} worker;

248/* ── Threadpool ──────────────────────────────────────────────────────────── */
249/*
250 * shutdown set by destroy(); workers exit their loop after observing
251 * it. seq_cst store, acquire reads.
252 * num_threads_alive workers decrement on exit; destroy spins until zero.
253 * workers_ready startup barrier: each worker increments before entering
254 * its main loop and spins until the count reaches
255 * num_workers, guaranteeing no worker steals from a
256 * half-initialised workers[] (Bug #3).
257 * num_pending outstanding-task counter — see ownership invariant in
258 * the file header. Sole idle predicate.
259 */
260struct Threadpool {
261 CACHE_ALIGNED atomic_int shutdown;
262 CACHE_ALIGNED atomic_int num_threads_alive;
263
264 CACHE_ALIGNED worker** workers;
265 CACHE_ALIGNED size_t num_workers;
266
267 /* Startup barrier */
268 CACHE_ALIGNED atomic_size_t workers_ready;
269
270 CACHE_ALIGNED GlobalQueue gq;
271
272 /* Parking */
273 CACHE_ALIGNED Lock park_lock;
274 CACHE_ALIGNED atomic_int num_parked;
275 Condition work_available;
276
277 /* Idle detection */
278 CACHE_ALIGNED Lock idle_lock;
279 CACHE_ALIGNED atomic_size_t num_pending;
280 Condition all_idle;
281};

313static bool gq_push(GlobalQueue* gq, Task task) {
314 lock_acquire(&gq->mutex);
315
316 while (((atomic_load_explicit(&gq->head, memory_order_relaxed) + 1) & GLOBAL_Q_MASK) ==
317 atomic_load_explicit(&gq->tail, memory_order_relaxed)) {
318 if (atomic_load_explicit(&gq->pool->shutdown, memory_order_acquire)) {
319 lock_release(&gq->mutex);
320 return false;
321 }
322 unpark_all(gq->pool); /* Bug #7: wake consumers before sleeping full */
323 cond_wait(&gq->not_full, &gq->mutex);
324 }
325
326 uint32_t h = atomic_load_explicit(&gq->head, memory_order_relaxed);
327 gq->tasks[h] = task;
328 atomic_store_explicit(&gq->head, (h + 1) & GLOBAL_Q_MASK, memory_order_release);
329 lock_release(&gq->mutex);
330 return true;
331}

338static size_t gq_push_batch_direct(GlobalQueue* gq, void (**functions)(void*), void** args, size_t count) {
339 size_t done = 0; /* caller-array slots consumed */
340 size_t pushed = 0; /* non-NULL tasks actually enqueued (returned) */
341
342 while (done < count) {
343 /* Compact leading NULLs out of the remaining window first. */
344 while (done < count && !functions[done]) done++;
345 if (done >= count) break;
346
347 lock_acquire(&gq->mutex);
348
349 while (((atomic_load_explicit(&gq->head, memory_order_relaxed) + 1) & GLOBAL_Q_MASK) ==
350 atomic_load_explicit(&gq->tail, memory_order_relaxed)) {
351 if (atomic_load_explicit(&gq->pool->shutdown, memory_order_acquire)) {
352 lock_release(&gq->mutex);
353 return pushed;
354 }
355 /*
356 * Bug #7 (deadlock): we are about to sleep on a FULL queue, but
357 * consumers are only signalled AFTER submit_batch() finishes
358 * streaming. A batch larger than the queue therefore deadlocks
359 * here whenever every worker managed to park before this point:
360 * we wait for a drain that nobody was woken to perform.
361 * Wake consumers BEFORE sleeping — they will drain the queue
362 * and broadcast not_full back to us.
363 */
364 unpark_all(gq->pool);
365 cond_wait(&gq->not_full, &gq->mutex);
366 }
367
368 uint32_t h = atomic_load_explicit(&gq->head, memory_order_relaxed);
369 uint32_t t = atomic_load_explicit(&gq->tail, memory_order_relaxed);
370 size_t free_slots = (GLOBAL_Q_SIZE - 1) - ((h - t) & GLOBAL_Q_MASK);
371
372 /* Pack as many non-NULL entries as fit in one hold of the lock.
373 * At least one fits: functions[done] is non-NULL and the queue
374 * has room, so progress per iteration is guaranteed. */
375 size_t i = done;
376 size_t n = 0;
377 while (i < count && n < free_slots) {
378 if (functions[i]) {
379 gq->tasks[(h + n) & GLOBAL_Q_MASK] = (Task){functions[i], args ? args[i] : NULL};
380 n++;
381 }
382 i++;
383 }
384
385 atomic_store_explicit(&gq->head, (h + (uint32_t)n) & GLOBAL_Q_MASK, memory_order_release);
386 pushed += n;
387 done = i;
388
389 lock_release(&gq->mutex);
390 }
391
392 return pushed;
393}

404static int gq_pull_batch(GlobalQueue* gq, Task* out, size_t max) {
405 lock_acquire(&gq->mutex);
406
407 uint32_t h = atomic_load_explicit(&gq->head, memory_order_relaxed);
408 uint32_t t = atomic_load_explicit(&gq->tail, memory_order_relaxed);
409
410 if (h == t) {
411 lock_release(&gq->mutex);
412 return atomic_load_explicit(&gq->pool->shutdown, memory_order_acquire) ? -1 : 0;
413 }
414
415 size_t available = (h - t) & GLOBAL_Q_MASK;
416 size_t to_take = available < max ? available : max;
417
418 for (size_t i = 0; i < to_take; i++) {
419 out[i] = gq->tasks[(t + i) & GLOBAL_Q_MASK];
420 }
421 atomic_store_explicit(&gq->tail, (t + (uint32_t)to_take) & GLOBAL_Q_MASK, memory_order_release);
422
423 cond_broadcast(&gq->not_full);
424 lock_release(&gq->mutex);
425 return (int)to_take;
426}

444static inline bool deque_push_bottom(WorkStealDeque* dq, Task task) {
445 size_t b = atomic_load_explicit(&dq->bottom, memory_order_relaxed);
446 size_t t = atomic_load_explicit(&dq->top, memory_order_acquire);
447
448 if (b - t >= DEQUE_SIZE) return false;
449
450 dq->tasks[b & DEQUE_MASK] = task;
451 atomic_store_explicit(&dq->bottom, b + 1, memory_order_release);
452 return true;
453}

467static inline bool deque_pop_bottom(WorkStealDeque* dq, Task* out) {
468 size_t b = atomic_load_explicit(&dq->bottom, memory_order_relaxed);
469 size_t t = atomic_load_explicit(&dq->top, memory_order_acquire);
470
471 if ((ptrdiff_t)(b - t) <= 0) return false;
472
473 b--;
474 atomic_store_explicit(&dq->bottom, b, memory_order_seq_cst);
475 t = atomic_load_explicit(&dq->top, memory_order_seq_cst);
476
477 if ((ptrdiff_t)(b - t) > 0) {
478 *out = dq->tasks[b & DEQUE_MASK];
479 return true;
480 }
481
482 if (b == t) {
483 *out = dq->tasks[b & DEQUE_MASK];
484 size_t expected = t;
485 bool won = atomic_compare_exchange_strong_explicit(&dq->top, &expected, t + 1, memory_order_seq_cst,
486 memory_order_relaxed);
487 atomic_store_explicit(&dq->bottom, b + 1, memory_order_relaxed);
488 return won;
489 }
490
491 atomic_store_explicit(&dq->bottom, b + 1, memory_order_relaxed);
492 return false;
493}

502static inline StealResult deque_steal_top(WorkStealDeque* dq, Task* out) {
503 size_t t = atomic_load_explicit(&dq->top, memory_order_acquire);
504 size_t b = atomic_load_explicit(&dq->bottom, memory_order_acquire);
505
506 if ((ptrdiff_t)(b - t) <= 0) return STEAL_EMPTY;
507
508 *out = dq->tasks[t & DEQUE_MASK];
509
510 if (!atomic_compare_exchange_strong_explicit(&dq->top, &t, t + 1, memory_order_seq_cst, memory_order_relaxed)) {
511 return STEAL_ABORT;
512 }
513 return STEAL_SUCCESS;
514}

520static void worker_park(Threadpool* pool) {
521 /*
522 * Lost-wakeup proof (Bug #6): the four ops below — our parked++ and
523 * pending check here, plus the submitter's pending++ and parked read
524 * in unpark_*() — must be seq_cst so they share one total order.
525 *
526 * Suppose our pending check reads 0 (pre-increment value of a racing
527 * submitter). Seq_cst program order then forces:
528 * parked++ < pending-check < submitter's pending++ < its parked read,
529 * so the submitter MUST observe num_parked > 0 and signal us.
530 * Relaxed/acquire on either side allows both sides to see stale zeros
531 * and the worker would sleep through work that already arrived.
532 *
533 * On x86 this costs nothing: seq_cst loads compile to plain MOV and
534 * the RMW was already a LOCK'd instruction.
535 */
536 atomic_fetch_add_explicit(&pool->num_parked, 1, memory_order_seq_cst);
537 lock_acquire(&pool->park_lock);
538
539 /* Direct pending check avoids scanning every worker's deque headers */
540 bool has_work = (atomic_load_explicit(&pool->num_pending, memory_order_seq_cst) > 0);
541
542 if (!has_work && !atomic_load_explicit(&pool->shutdown, memory_order_acquire)) {
543 cond_wait(&pool->work_available, &pool->park_lock);
544 }
545
546 lock_release(&pool->park_lock);
547 atomic_fetch_sub_explicit(&pool->num_parked, 1, memory_order_relaxed);
548}

550static inline void unpark_one(Threadpool* pool) {
551 if (atomic_load_explicit(&pool->num_parked, memory_order_seq_cst) > 0) {
552 lock_acquire(&pool->park_lock);
553 cond_signal(&pool->work_available);
554 lock_release(&pool->park_lock);
555 }
556}

572static inline void unpark_n(Threadpool* pool, size_t n) {
573 int parked = atomic_load_explicit(&pool->num_parked, memory_order_seq_cst);
574 if (parked <= 0) return;
575
576 lock_acquire(&pool->park_lock);
577 if ((size_t)parked <= n && parked > 1) {
578 cond_broadcast(&pool->work_available);
579 } else {
580 if ((size_t)parked > n) parked = (int)n;
581 while (parked-- > 0) {
582 cond_signal(&pool->work_available);
583 }
584 }
585 lock_release(&pool->park_lock);
586}

588static inline void unpark_all(Threadpool* pool) {
589 lock_acquire(&pool->park_lock);
590 cond_broadcast(&pool->work_available);
591 lock_release(&pool->park_lock);
592}

617static bool try_steal(worker* self, Task* out) {
618 Threadpool* pool = self->pool;
619 size_t n = pool->num_workers;
620
621 static _Thread_local uint64_t rng = 0;
622 if (rng == 0) rng = ((uint64_t)self->index + 1) * 6364136223846793005ULL;
623 rng ^= rng << 13;
624 rng ^= rng >> 7;
625 rng ^= rng << 17;
626
627 /* Lemire's fast reduction replaces integer division with multiply+shift */
628 size_t start = (size_t)(((uint64_t)(uint32_t)rng * (uint64_t)n) >> 32);
629
630 /* 1. Lock-free steal from private deques */
631 size_t v = start;
632 for (size_t i = 0; i < n; i++) {
633 if (v != self->index) {
634 StealResult r = deque_steal_top(&pool->workers[v]->deque, out);
635 if (r == STEAL_SUCCESS) return true;
636 }
637 if (++v >= n) v = 0;
638 }
639
640 /* 2. Batch drain from global queue */
641 Task batch[BATCH_SIZE];
642 int got = gq_pull_batch(&pool->gq, batch, BATCH_SIZE);
643 if (got <= 0) return false;
644
645 unpark_one(pool);
646
647 for (int i = 1; i < got; i++) {
648 if (!deque_push_bottom(&self->deque, batch[i])) {
649 for (int j = i; j < got; j++) {
650 gq_push(&pool->gq, batch[j]);
651 }
652 break;
653 }
654 }
655
656 *out = batch[0];
657 return true;
658}

681static void* worker_thread(void* arg) {
682 worker* self = (worker*)arg;
683 Threadpool* pool = self->pool;
684 Task task;
685 int spin = 0;
686 size_t local_done = 0;
687
688 tls_worker_index = self->index;
689
690 atomic_fetch_add_explicit(&pool->workers_ready, 1, memory_order_release);
691 while (atomic_load_explicit(&pool->workers_ready, memory_order_acquire) < pool->num_workers) {
692 thread_yield();
693 }
694
695 atomic_fetch_add_explicit(&pool->num_threads_alive, 1, memory_order_relaxed);
696
697 while (!atomic_load_explicit(&pool->shutdown, memory_order_acquire)) {
698 if (deque_pop_bottom(&self->deque, &task)) goto execute;
699
700 if (try_steal(self, &task)) goto execute;
701
702 /* Drain point: publish completions so wait()/destroy() never lag */
703 if (local_done) {
704 size_t done = local_done;
705 local_done = 0;
706 size_t pending = atomic_fetch_sub_explicit(&pool->num_pending, done, memory_order_acq_rel) - done;
707 if (pending == 0) {
708 lock_acquire(&pool->idle_lock);
709 if (atomic_load_explicit(&pool->num_pending, memory_order_relaxed) == 0) {
710 cond_broadcast(&pool->all_idle);
711 }
712 lock_release(&pool->idle_lock);
713 }
714 }
715
716 /*
717 * Exponential PAUSE backoff, then park (Perf #9).
718 *
719 * An earlier tier called sched_yield() between pauses and parking;
720 * strace showed ~16 yield syscalls per submitted task with a ~45us
721 * average service time — losing threads were paying kernel time
722 * slices for the privilege of trying again almost immediately.
723 * Pure pause-spinning bounded by YIELD_THRESHOLD followed by a
724 * proper park gives the same latency benefit at a fraction of the
725 * system cost.
726 */
727 if (spin < YIELD_THRESHOLD) {
728 spin++;
729 int pauses = 1 << spin;
730 for (int k = 0; k < pauses; k++) {
731 cpu_relax();
732 }
733 continue;
734 }
735
736 spin = 0;
737 worker_park(pool);
738 continue;
739
740 execute:
741 spin = 0;
742 task.function(task.arg);
743
744 if (++local_done >= PENDING_FLUSH_BATCH) {
745 size_t done = local_done;
746 local_done = 0;
747 size_t pending = atomic_fetch_sub_explicit(&pool->num_pending, done, memory_order_acq_rel) - done;
748 if (pending == 0) {
749 lock_acquire(&pool->idle_lock);
750 if (atomic_load_explicit(&pool->num_pending, memory_order_relaxed) == 0) {
751 cond_broadcast(&pool->all_idle);
752 }
753 lock_release(&pool->idle_lock);
754 }
755 }
756 }
757
758 if (local_done) {
759 atomic_fetch_sub_explicit(&pool->num_pending, local_done, memory_order_acq_rel);
760 }
761
762 int alive = atomic_fetch_sub_explicit(&pool->num_threads_alive, 1, memory_order_acq_rel) - 1;
763 if (alive == 0) {
764 lock_acquire(&pool->idle_lock);
765 cond_broadcast(&pool->all_idle);
766 lock_release(&pool->idle_lock);
767 }
768
769 return NULL;
770}

800Threadpool* threadpool_create(size_t num_threads) {

802
803 Threadpool* pool = (Threadpool*)ALIGNED_ALLOC(CACHE_LINE_SIZE, sizeof(Threadpool));
804 if (!pool) return NULL;
805
806 atomic_store_explicit(&pool->shutdown, 0, memory_order_relaxed);
807 atomic_store_explicit(&pool->num_threads_alive, 0, memory_order_relaxed);
808 atomic_store_explicit(&pool->num_parked, 0, memory_order_relaxed);
809 atomic_store_explicit(&pool->num_pending, 0, memory_order_relaxed);
810 atomic_store_explicit(&pool->workers_ready, 0, memory_order_relaxed);
811
812 pool->num_workers = num_threads;
813 gq_init(&pool->gq, pool);
814
815 lock_init(&pool->park_lock);
816 cond_init(&pool->work_available);
817 lock_init(&pool->idle_lock);
818 cond_init(&pool->all_idle);
819
820 pool->workers = (worker**)malloc(num_threads * sizeof(worker*));
821 if (!pool->workers) {
822 gq_destroy(&pool->gq);
823 lock_free(&pool->park_lock);
824 cond_free(&pool->work_available);
825 lock_free(&pool->idle_lock);
826 cond_free(&pool->all_idle);
827 free(pool);
828 return NULL;
829 }
830
831 for (size_t i = 0; i < num_threads; i++) pool->workers[i] = NULL;
832
833 for (size_t i = 0; i < num_threads; i++) {
834 if (worker_init(pool, &pool->workers[i], i) != 0) {
835 atomic_store_explicit(&pool->workers_ready, num_threads, memory_order_release);
836 pool->num_workers = i;
837 threadpool_destroy(pool, -1);
838 return NULL;
839 }
840 }

842 return pool;

853bool threadpool_submit(Threadpool* pool, void (*function)(void*), void* arg) {
854 if (!pool || !function) return false;
855
856 Task task = {function, arg};
857 atomic_fetch_add_explicit(&pool->num_pending, 1, memory_order_seq_cst);
858
859 if (tls_worker_index != SIZE_MAX) {
860 if (deque_push_bottom(&pool->workers[tls_worker_index]->deque, task)) {
861 unpark_one(pool);
862 return true;
863 }
864 }
865
866 bool ok = gq_push(&pool->gq, task);
867 if (ok) {
868 unpark_one(pool);
869 } else {
870 atomic_fetch_sub_explicit(&pool->num_pending, 1, memory_order_relaxed);
871 }
872 return ok;
873}

891size_t threadpool_submit_batch(Threadpool* pool, void (**functions)(void*), void** args, size_t count) {
892 if (!pool || !functions || count == 0) return 0;
893
894 if (tls_worker_index != SIZE_MAX) {
895 atomic_fetch_add_explicit(&pool->num_pending, count, memory_order_seq_cst);
896
897 size_t pushed = 0;
898 for (size_t i = 0; i < count; i++) {
899 if (!functions[i]) continue;
900 Task task = {functions[i], args ? args[i] : NULL};
901 if (deque_push_bottom(&pool->workers[tls_worker_index]->deque, task)) {
902 pushed++;
903 } else {
904 pushed += gq_push_batch_direct(&pool->gq, &functions[i], args ? &args[i] : NULL, count - i);
905 break;
906 }
907 }
908 if (pushed < count) {
909 atomic_fetch_sub_explicit(&pool->num_pending, count - pushed, memory_order_relaxed);
910 }
911 if (pushed > 0) unpark_n(pool, pushed / BATCH_SIZE + 1);
912 return pushed;
913 }
914
915 /* External thread: Zero-alloc direct stream into global queue */
916 atomic_fetch_add_explicit(&pool->num_pending, count, memory_order_seq_cst);
917 size_t pushed = gq_push_batch_direct(&pool->gq, functions, args, count);
918 if (pushed < count) {
919 atomic_fetch_sub_explicit(&pool->num_pending, count - pushed, memory_order_relaxed);
920 }
921
922 if (pushed > 0) unpark_n(pool, pushed / BATCH_SIZE + 1);
923 return pushed;
924}

926/*
927 * Adaptive wait threshold (Perf #8): after a submit, the last task is
928 * typically finished within a few microseconds. Sleeping in threadpool_wait()
929 * costs two futex transitions (block + wake) plus an idle broadcast from the
930 * completing worker — measurable in sub-millisecond submit/wait cycles
931 * (perf: >60% of wall time was sys-time in ping-pong workloads).
932 *
933 * Before blocking we therefore poll num_pending with PAUSE-backed spins for
934 * roughly ADAPTIVE_SPIN_NS nanoseconds. If the pool drains inside that
935 * window we return without ANY syscall; otherwise we fall through to the
936 * cond_wait slow path unchanged.
937 */
938#define ADAPTIVE_SPIN_NS 4000
939
955void threadpool_wait(Threadpool* pool) {
956 if (!pool) return;
957
958 if (atomic_load_explicit(&pool->num_pending, memory_order_acquire) == 0) {
959 return;
960 }
961
962 uint64_t spins = ADAPTIVE_SPIN_NS / 20; /* ~20ns per pause+load pair */
963 do {
964 if (atomic_load_explicit(&pool->num_pending, memory_order_acquire) == 0) {
965 return;
966 }
967 cpu_relax();
968 } while (--spins);
969
970 lock_acquire(&pool->idle_lock);
971 while (atomic_load_explicit(&pool->num_pending, memory_order_relaxed) != 0) {
972 cond_wait(&pool->all_idle, &pool->idle_lock);
973 }
974 lock_release(&pool->idle_lock);
975}

995void threadpool_destroy(Threadpool* pool, int timeout_ms) {
996 if (!pool) return;
997
998 if (timeout_ms >= 0) {
999 lock_acquire(&pool->idle_lock);
1000 while (atomic_load_explicit(&pool->num_pending, memory_order_relaxed) != 0) {
1001 int r = cond_wait_timeout(&pool->all_idle, &pool->idle_lock, timeout_ms);
1002 if (r != 0) break;
1003 }
1004 lock_release(&pool->idle_lock);
1005 } else {
1006 lock_acquire(&pool->idle_lock);
1007 while (atomic_load_explicit(&pool->num_pending, memory_order_relaxed) != 0) {
1008 cond_wait(&pool->all_idle, &pool->idle_lock);
1009 }
1010 lock_release(&pool->idle_lock);
1011 }
1012
1013 atomic_store_explicit(&pool->shutdown, 1, memory_order_seq_cst);
1014
1015 unpark_all(pool);
1016
1017 while (atomic_load_explicit(&pool->num_threads_alive, memory_order_acquire) > 0) {
1018 unpark_all(pool);
1019 thread_yield();
1020 }
1021
1022 for (size_t i = 0; i < pool->num_workers; i++) {
1023 if (pool->workers[i]) {
1024 thread_join(pool->workers[i]->pthread, NULL);
1025 free(pool->workers[i]);
1026 }
1027 }
1028 free(pool->workers);
1029
1030 gq_destroy(&pool->gq);
1031 lock_free(&pool->park_lock);
1032 cond_free(&pool->work_available);
1033 lock_free(&pool->idle_lock);
1034 cond_free(&pool->all_idle);
1035 free(pool);
1036}

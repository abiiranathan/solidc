30/* -------------------------------------------------------------------------
31 * Thread-local overflow-block recycle bin
32 *
33 * Problem
34 * -------
35 * Short-lived arenas that outgrow their first block pay a heavy tax on
36 * every create/destroy cycle: overflow blocks are multi-page allocations
37 * which the system allocator satisfies with mmap() and returns to the OS
38 * on free(). The next arena then faults every page back in. In the
39 * bundled benchmark (bench_arena, Large scenario) this made cold arenas
40 * ~170x slower than warm ones.
41 *
42 * Mechanism
43 * ---------
44 * arena_destroy() hands its overflow blocks to a per-thread LIFO recycle
45 * bin instead of free(); _arena_alloc_slow() reuses a cached slab (best
46 * fit) before calling aligned_alloc_xp(). Pages therefore stay mapped
47 * and resident: no syscall, no page faults.
48 *
49 * Bounds & hygiene
50 * ----------------
51 * The bin is capped both in block count and total bytes; excess blocks go
52 * straight to free(). The cache is drained by a TLS destructor when a
53 * thread exits, so nothing is ever leaked. Sanitizer builds disable the
54 * bin entirely for precise leak attribution. Define
55 * ARENA_BLOCK_CACHE_COUNT=0 to turn recycling off at compile time.
56 * ---------------------------------------------------------------------- */
57

162static char* blk_cache_take(size_t min_slab, size_t* slab_out) {
163 if (ARENA_UNLIKELY(blk_cache_count == 0)) return NULL;
164
165 size_t best = SIZE_MAX;
166 size_t best_size = SIZE_MAX;
167 for (size_t i = 0; i < blk_cache_count; i++) {
168 if (blk_cache_slab[i] >= min_slab && blk_cache_slab[i] < best_size) {
169 best = i;
170 best_size = blk_cache_slab[i];
171 }
172 }
173 if (best == SIZE_MAX) return NULL;
174
175 char* ptr = (char*)blk_cache[best];
176 blk_cache_bytes -= blk_cache_slab[best];
177 blk_cache[best] = blk_cache[--blk_cache_count];
178 blk_cache_slab[best] = blk_cache_slab[blk_cache_count];
179 *slab_out = best_size;
180 return ptr;
181}

184static void blk_cache_put(ArenaBlock* block, size_t slab_size) {
185 static THREAD_LOCAL bool registered = false;
186 if (ARENA_UNLIKELY(!registered)) {
187 blk_cache_register();
188 registered = true;
189 }
190
191 if (blk_cache_count >= ARENA_BLOCK_CACHE_COUNT ||
192 blk_cache_bytes + slab_size > ARENA_BLOCK_CACHE_BYTES) {
193 aligned_free_xp(block);
194 return;
195 }
196 blk_cache[blk_cache_count] = block;
197 blk_cache_slab[blk_cache_count] = slab_size;
198 blk_cache_count++;
199 blk_cache_bytes += slab_size;
200}

269Arena* arena_create(size_t reserve_size) {
270 Arena* a = (Arena*)aligned_alloc_xp(64, sizeof(Arena));
271#ifdef ARENA_ABORT_ON_OOM
272 if (ARENA_UNLIKELY(!a)) ARENA_OOM_HANDLER(sizeof(Arena));
273#endif
274 if (!a) return NULL;
275
276 if (!static_buffer_in_use && reserve_size <= STATIC_BUFFER_SIZE) {
277 arena_init(a, static_buffer, STATIC_BUFFER_SIZE);
278 static_buffer_in_use = true;
279
280 // Avoid race condition if the arena is handed off to another thread
281 // after creation since we are storing this threads address.
282 a->tls_in_use_origin = &static_buffer_in_use;
283 } else {
284 size_t page_size = get_page_size();
285 size_t initial_size = reserve_size > 0 ? reserve_size : ARENA_MIN_BLOCK_SIZE;
286 /* Round up to page boundary. */
287 initial_size = (initial_size + page_size - 1) & ~(page_size - 1);
288
289 char* buf = (char*)aligned_alloc_xp(64, initial_size);
290 if (!buf) {
291 aligned_free_xp(a);
292#ifdef ARENA_ABORT_ON_OOM
293 ARENA_OOM_HANDLER(initial_size);
294 ARENA_UNREACHABLE(); /* handler aborts; silences MSVC C4702 */
295#else
296 return NULL;
297#endif
298 }
299 arena_init(a, buf, initial_size);
300 /* Mark the first block as heap-owned so arena_destroy frees it. */
301 a->first_block.is_static = false;
302 }
303
304 a->heap_allocated = true;
305 return a;
306}

312void arena_destroy(Arena* a) {
313 if (!a) return;
314
315 ArenaBlock* block = a->head;
316
317 /* Free the first block's buffer only if we heap-allocated it.
318 * Static/stack/TLS-backed buffers are caller-owned and must not be freed. */
319 if (block && !block->is_static) {
320 aligned_free_xp(block->base);
321 }
322
323 /* If the first block was the TLS static buffer, mark it available again
324 * so the next arena_create() can reuse it without leaking capacity.
325 Comparing block->base == static_buffer is a bug if the arena switched threads.
326 */
327 if (block && block->is_static && a->tls_in_use_origin) {
328 *a->tls_in_use_origin = false;
329 }
330
331 /* Walk overflow blocks. Each was allocated as a single slab where the
332 * ArenaBlock header lives at the start of the pointer returned by
333 * aligned_alloc_xp, so we free the block pointer itself (not block->base).
334 * Blocks are offered to the thread-local recycle bin first so a future
335 * arena can reuse them without syscalls or page faults. */
336 block = block ? block->next : NULL;
337 while (block) {
338 ArenaBlock* next = block->next;
339 blk_cache_put(block, (size_t)(block->end - (char*)block));
340 block = next;
341 }
342
343 if (a->heap_allocated) aligned_free_xp(a);
344}

355void* _arena_alloc_slow(Arena* a, size_t size, size_t alignment) {
356 /* --- Try existing cached blocks after current_block first ------------ */
357
358 ArenaBlock* next = a->current_block->next;
359 while (next) {
360 uintptr_t aligned = ((uintptr_t)next->base + alignment - 1) & ~(uintptr_t)(alignment - 1);
361 if (aligned + size <= (uintptr_t)next->end) {
362 /* Found a cached block with enough room. */
363 a->current_block = next;
364 a->curr = (char*)(aligned + size);
365 a->end = next->end;
366 return (void*)aligned;
367 }
368 next = next->next;
369 }
370
371 /* --- Allocate a fresh block ----------------------------------------- */
372
373 /* The new block slab must hold: ArenaBlock header + worst-case alignment
374 * padding + the requested allocation. */
375 size_t current_capacity = (size_t)(a->current_block->end - a->current_block->base);
376 size_t needed = sizeof(ArenaBlock) + (alignment - 1) + size;
377
378 // Check for overflow.
379 size_t next_size = (current_capacity <= ARENA_MAX_BLOCK_SIZE / 2) ? current_capacity * 2
380 : ARENA_MAX_BLOCK_SIZE;
381 if (next_size < needed) next_size = needed;
382 if (next_size < ARENA_MIN_BLOCK_SIZE) next_size = ARENA_MIN_BLOCK_SIZE;
383
384 /* Round up to page boundary so mmap-based backends work efficiently. */
385 next_size = (next_size + a->page_size - 1) & ~(a->page_size - 1);
386
387 /* Recycle bin first: reusing a cached slab avoids the allocator call
388 * and, more importantly, keeps pages mapped so no page faults occur. */
389 size_t got = 0;
390 char* ptr = blk_cache_take(next_size, &got);
391 if (ptr) {
392 next_size = got; /* adopt the full slab so no capacity is stranded */
393 } else {
394 ptr = (char*)aligned_alloc_xp(64, next_size);
395 }
396 if (!ptr) return NULL;
397
398 ArenaBlock* block = (ArenaBlock*)ptr;
399
400 /* base starts right after the header, already accounting for the 64-byte
401 * slab alignment from aligned_alloc_xp. If the requested alignment is
402 * <= 64 (the common case) block->base is already correctly aligned — the
403 * extra alignment arithmetic below reduces to a no-op. */
404 uintptr_t base_addr = (uintptr_t)(ptr + sizeof(ArenaBlock));
405 uintptr_t aligned = (base_addr + alignment - 1) & ~(uintptr_t)(alignment - 1);
406
407 block->base = (char*)base_addr; /* record the true start for capacity math */
408 block->end = ptr + next_size;
409 block->is_static = false;
410 /* Insert at the tail of the chain so skipped cached blocks are revisited
411 * on future slow-path calls instead of being orphaned mid-chain. */
412 block->next = NULL;
413 a->current_block->next = block;
414
415 a->current_block = block;
416 a->total_committed += next_size;
417 a->curr = (char*)(aligned + size);
418 a->end = block->end;
419
420 return (void*)aligned;
421}

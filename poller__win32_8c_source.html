52static int poller_rebuild(Poller* p) {
53 if (p->fd_capacity < p->base.regs.count) {
54 size_t new_cap = p->base.regs.count < 16 ? 16 : p->base.regs.count * 2;
55 WSAPOLLFD* nf = (WSAPOLLFD*)realloc(p->fds, new_cap * sizeof(WSAPOLLFD));
56 if (!nf) {
57 errno = ENOMEM;
58 return -1;
59 }
60 p->fds = nf;
61 void** nd = (void**)realloc(p->datas, new_cap * sizeof(void*));
62 if (!nd) {
63 errno = ENOMEM;
64 return -1;
65 }
66 p->datas = nd;
67 p->fd_capacity = new_cap;
68 }
69
70 size_t n = 0;
71 for (size_t i = 0; i < p->base.regs.capacity; i++) {
72 PollerReg* r = &p->base.regs.slots[i];
73 if (!r->active) continue;
74 p->fds[n].fd = r->fd;
75 p->fds[n].events = (short)((r->events & POLLER_READ) ? (POLLRDNORM | POLLRDBAND) : 0);
76 if (r->events & POLLER_WRITE) p->fds[n].events |= POLLWRNORM;
77 p->fds[n].revents = 0;
78 p->datas[n] = r->data;
79 n++;
80 }
81 p->fd_count = n;
82 return 0;
83}

123int poller_wait(Poller* p, PollerEvent* events, int max, int timeout_ms) {
124 if (!p || !events || max <= 0) {
125 errno = EINVAL;
126 return -1;
127 }
128 if (p->fd_count == 0) {
129 if (timeout_ms > 0) Sleep((DWORD)timeout_ms);
130 return 0;
131 }
132
133 int n = WSAPoll(p->fds, (ULONG)p->fd_count, timeout_ms);
134 if (n <= 0) return n;
135
136 int out = 0;
137 for (size_t i = 0; i < p->fd_count && out < max; i++) {
138 SHORT re = p->fds[i].revents;
139 if (!re) continue;
140
141 events[out].fd = (int)p->fds[i].fd;
142 events[out].data = p->datas[i];
143 events[out].readable = (re & (POLLRDNORM | POLLRDBAND)) != 0;
144 events[out].writable = (re & POLLWRNORM) != 0;
145 events[out].error = (re & POLLERR) != 0;
146 events[out].hup = (re & POLLHUP) != 0; /* WSAPoll has no RDHUP */
147 out++;
148 }
149 return out;
150}

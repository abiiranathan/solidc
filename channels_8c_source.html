47Channel* chan_new(size_t value_size, size_t capacity_hint) {
48 if (value_size == 0) return NULL;
49
50 Channel* c = (Channel*)calloc(1, sizeof(Channel));
51 if (!c) return NULL;
52
53 c->value_size = value_size;
54 c->capacity = chan_pow2(capacity_hint ? capacity_hint : 16);
55 c->buf = (unsigned char*)malloc(c->capacity * value_size);

79static ChanStatus chan_grow(Channel* c, size_t new_cap) {
80 unsigned char* nb = (unsigned char*)malloc(new_cap * c->value_size);
81 if (!nb) return CHAN_NOMEM;
82
83 /* Re-linearize: copy head..wrap into the front of the new buffer. */
84 size_t tail_len = c->capacity - c->head;
85 size_t first = (c->count < tail_len) ? c->count : tail_len;
86 memcpy(nb, c->buf + c->head * c->value_size, first * c->value_size);
87 if (c->count > first) {
88 memcpy(nb + first * c->value_size, c->buf, (c->count - first) * c->value_size);
89 }
90
91 free(c->buf);
92 c->buf = nb;
93 c->capacity = new_cap;
94 c->head = 0;
95 return CHAN_OK;
96}

98ChanStatus chan_send(Channel* c, const void* value) {
99 if (!c || !value) return CHAN_INVALID;
100
101 ChanStatus st = CHAN_OK;
102 lock_acquire(&c->lock);
103 for (;;) {
104 if (c->closed) {
105 st = CHAN_CLOSED;
106 goto out;
107 }
108 if (c->count < c->capacity) break;
109
110 /* Grow unbounded (Go-like): senders never block on a full buffer. */
111 st = chan_grow(c, c->capacity * 2);
112 if (st != CHAN_OK) goto out;
113 break;
114 }
115
116 size_t tail = (c->head + c->count) & (c->capacity - 1);
117 memcpy(c->buf + tail * c->value_size, value, c->value_size);
118 c->count++;
119 cond_signal(&c->not_empty);
120 st = CHAN_OK;
121
122out:
123 lock_release(&c->lock);
124 return st;
125}

167ChanStatus chan_recv_timeout(Channel* c, void* out, int timeout_ms) {
168 if (!c || !out) return CHAN_INVALID;
169
170 ChanStatus st = CHAN_OK;
171 lock_acquire(&c->lock);
172 for (;;) {
173 if (c->count > 0) {
174 memcpy(out, c->buf + c->head * c->value_size, c->value_size);
175 c->head = (c->head + 1) & (c->capacity - 1);
176 c->count--;
177 cond_signal(&c->not_full);
178 st = CHAN_OK;
179 break;
180 }
181 if (c->closed) {
182 st = CHAN_CLOSED;
183 break;
184 }
185 if (timeout_ms == 0) {
186 st = CHAN_TIMEOUT;
187 break;
188 }
189 if (cond_wait_timeout(&c->not_empty, &c->lock, timeout_ms) != 0) {
190 /* Timed out (or rare spurious failure): re-check state once so a
191 * value racing with the timeout is still delivered. */
192 if (c->count > 0) continue;
193 st = CHAN_TIMEOUT;
194 break;
195 }
196 /* Woken: loop re-checks count/closed. */
197 }
198 lock_release(&c->lock);
199 return st;
200}

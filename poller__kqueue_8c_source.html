63static int kev_add(struct kevent* out, int kqfd, int fd, int16_t filter, bool enable, void* udata) {
64 if (enable) {
65 EV_SET(out, fd, filter, EV_ADD | EV_CLEAR, 0, 0, udata);
66 } else {
67 /* Disable (not delete) so the sibling filter stays intact. */
68 EV_SET(out, fd, filter, EV_DISABLE, 0, 0, udata);
69 }
70 return kevent(kqfd, out, 1, NULL, 0, NULL);
71}

74 if (!p || fd < 0) {

121int poller_del(Poller* p, int fd) {
122 if (!p || fd < 0) {
123 errno = EINVAL;
124 return -1;
125 }
126
127 struct kevent ev;
128 /* EV_DELETE on a filter that was never added returns ENOENT; treat as
129 * success so poller_del is forgiving like the epoll backend. */
130 EV_SET(&ev, fd, EVFILT_READ, EV_DELETE, 0, 0, NULL);
131 if (kevent(p->kqfd, &ev, 1, NULL, 0, NULL) != 0 && errno != ENOENT) return -1;
132 EV_SET(&ev, fd, EVFILT_WRITE, EV_DELETE, 0, 0, NULL);
133 if (kevent(p->kqfd, &ev, 1, NULL, 0, NULL) != 0 && errno != ENOENT) return -1;
134
135 poller_regs_remove(&p->base, fd);
136 return 0;
137}

139int poller_wait(Poller* p, PollerEvent* events, int max, int timeout_ms) {
140 if (!p || !events || max <= 0) {
141 errno = EINVAL;
142 return -1;
143 }
144
145 /* Grow the scratch buffer to the caller's batch size (never shrinks,
146 * so steady-state waits allocate nothing). */
147 if (max > p->kevs_cap) {
148 int cap = p->kevs_cap ? p->kevs_cap : 128;
149 while (cap < max) cap *= 2;
150 struct kevent* nkevs = (struct kevent*)realloc(p->kevs, (size_t)cap * sizeof(*nkevs));
151 if (!nkevs) {
152 errno = ENOMEM;
153 return -1;
154 }
155 p->kevs = nkevs;
156 p->kevs_cap = cap;
157 }
158
159 struct timespec ts, *ts_ptr = NULL;
160 if (timeout_ms >= 0) {
161 ts.tv_sec = timeout_ms / 1000;
162 ts.tv_nsec = (long)(timeout_ms % 1000) * 1000000L;
163 ts_ptr = &ts;
164 }
165
166 int n = kevent(p->kqfd, NULL, 0, p->kevs, max, ts_ptr);
167 if (n > 0) {
168 for (int i = 0; i < n; i++) {
169 events[i].fd = (int)p->kevs[i].ident;
170 events[i].data = p->kevs[i].udata;
171 events[i].readable = (p->kevs[i].filter == EVFILT_READ);
172 events[i].writable = (p->kevs[i].filter == EVFILT_WRITE);
173 events[i].error = (p->kevs[i].flags & EV_ERROR) != 0;
174 events[i].hup = (p->kevs[i].flags & EV_EOF) != 0;
175 }
176 }
177 return n;
178}
